    72→ImVec2 CtrlAbsPos(const Ctrl* c) {
    73→    int x = c->x, y = c->y;
    74→    for (Ctrl* p = c->parent; p; p = p->parent) { x += p->x; y += p->y; }
    75→    return ImVec2((float)x, (float)y);
    76→}

    79→//  form başlık bayrakları (üretilen kodla aynı set + editör sabitlemeleri)
    80→static ImGuiWindowFlags FormWinFlagsOf(const Model& m) {
    81→    ImGuiWindowFlags flags = ImGuiWindowFlags_NoSavedSettings;
    82→    auto it = m.formProps.find("Flags");
    83→    if (it != m.formProps.end())
    84→        for (auto& s : SplitW(Utf8ToWideStr(it->second), L'|')) {
    85→            std::string t = TrimA(WideStrToUtf8(s));
    86→            if (t == "NoTitleBar") flags |= ImGuiWindowFlags_NoTitleBar;
    87→            else if (t == "NoResize") flags |= ImGuiWindowFlags_NoResize;
    88→            else if (t == "NoMove") flags |= ImGuiWindowFlags_NoMove;
    89→            else if (t == "AlwaysAutoResize") flags |= ImGuiWindowFlags_AlwaysAutoResize;
    90→            else if (t == "NoScrollbar") flags |= ImGuiWindowFlags_NoScrollbar;
    91→            else if (t == "MenuBar") flags |= ImGuiWindowFlags_MenuBar;
    92→        }
    93→    for (Ctrl* c : m.ctrls) if (c->type == "Menu") { flags |= ImGuiWindowFlags_MenuBar; break; }
    94→    // editör sabitlemeleri (tasarım yüzeyinde form sabit kalır)
    95→    flags |= ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize |
    96→             ImGuiWindowFlags_NoFocusOnAppearing | ImGuiWindowFlags_NoBringToFrontOnFocus;
    97→    return flags;
    98→}

   102→static void PushRect(Ctrl* c, bool tray = false) {
   103→    WidgetRect r; r.c = c;
   104→    r.rmin = ImGui::GetItemRectMin();
   105→    r.rmax = ImGui::GetItemRectMax();
   106→    r.tray = tray;
   107→    s_hitRects.push_back(r);
   108→}
   109→// güvenli: widget çizilmediyse (0 boyut vb.) sağlam bir dikdörtgen kaydet
   110→static void PushRectFallback(Ctrl* c) {
   111→    WidgetRect r; r.c = c;
   112→    ImVec2 p = ImGui::GetCursorScreenPos();
   113→    r.rmin = p;
   114→    r.rmax = ImVec2(p.x + (float)(c->w ? c->w : 60), p.y + (float)(c->h ? c->h : 20));
   115→    s_hitRects.push_back(r);
   116→}

   126→static void DrawCtrl(Ctrl* c, Model& m) {
   127→    const std::string& t = c->type;
   128→    const ImVec2 pos((float)c->x, (float)c->y);
   129→    int stylePacked = objstyle::Push(c);
   130→    bool disabled = c->GetB("Disabled");
   131→    if (disabled) ImGui::BeginDisabled();
   132→
   133→    if (t == "Button") {
   134→        int w = atoi(c->Get("SizeArgW", "120").c_str());
   135→        int h = atoi(c->Get("SizeArgH", "28").c_str());
   136→        ImVec2 sz((float)(c->w > 0 ? c->w : w), (float)(c->h > 0 ? c->h : h));
   137→        ImGui::SetCursorPos(pos);
   138→        if (c->GetB("Small")) ImGui::SmallButton(WLabel(c, "Text", "Buton").c_str());
   139→        else ImGui::Button(WLabel(c, "Text", "Buton").c_str(), sz);
   140→        PushRect(c);
   141→    } else if (t == "Checkbox") {
   142→        bool v = c->GetB("Checked");
   143→        ImGui::SetCursorPos(pos);
   144→        ImGui::Checkbox(WLabel(c, "Text", "Seçenek").c_str(), &v);
   145→        PushRect(c);
   146→    } else if (t == "RadioButton") {
   147→        bool v = c->GetB("Active");
   148→        ImGui::SetCursorPos(pos);
   149→        ImGui::RadioButton(WLabel(c, "Text", "Seçim").c_str(), &v);
   150→        PushRect(c);
   151→    } else if (t == "SliderFloat" || t == "SliderInt") {
   152→        float fv = (float)atof(c->Get("Value", "0").c_str());
   153→        int   iv = atoi(c->Get("Value", "0").c_str());
   154→        float mn = (float)atof(c->Get("Min", "0").c_str());
   155→        float mx = (float)atof(c->Get("Max", "100").c_str());
   156→        const char* fmt = c->Get("Format", t == "SliderFloat" ? "%.1f" : "%d").c_str();
   157→        ImGui::SetCursorPos(pos);
   158→        if (t == "SliderFloat") ImGui::SliderFloat(WLabel(c, "Text", "Değer").c_str(), &fv, mn, mx, fmt);
   159→        else ImGui::SliderInt(WLabel(c, "Text", "Değer").c_str(), &iv, (int)mn, (int)mx, fmt);
   160→        PushRect(c);
   161→    } else if (t == "DragFloat" || t == "DragInt") {
   162→        float fv = (float)atof(c->Get("Value", "0").c_str());
   163→        int   iv = atoi(c->Get("Value", "0").c_str());
   164→        float mn = (float)atof(c->Get("Min", "0").c_str());
   165→        float mx = (float)atof(c->Get("Max", "100").c_str());
   166→        ImGui::SetCursorPos(pos);
   167→        if (t == "DragFloat") ImGui::DragFloat(WLabel(c, "Text", "Değer").c_str(), &fv, (float)atof(c->Get("Speed", "1").c_str()), mn, mx);
   168→        else ImGui::DragInt(WLabel(c, "Text", "Değer").c_str(), &iv, (float)atof(c->Get("Speed", "1").c_str()), (int)mn, (int)mx);
   169→        PushRect(c);
   170→    } else if (t == "InputText" || t == "InputTextMultiline") {
   171→        // her nesnenin KALICI kendi tamponu (Value'dan başlatılır)
   172→        std::string& buf = s_textBufs[c];
   173→        if (buf.empty() && !c->Get("Value", "").empty()) buf = c->Get("Value", "");
   174→        else if (buf.empty()) buf.resize(1, '\0');
   175→        // tasarımdaki Value değişti mi? (props panelinden) — senkron
   176→        static std::map<const Ctrl*, std::string> lastVal;
   177→        std::string curVal = c->Get("Value", "");
   178→        if (lastVal[c] != curVal) { buf = curVal; lastVal[c] = curVal; }
   179→        if (buf.size() < 512) buf.resize(512, '\0'); else buf.resize(buf.size() + 1, '\0');
   180→        ImGuiInputTextFlags fl = ImGuiInputTextFlags_ReadOnly;
   181→        if (c->GetB("Password")) fl |= ImGuiInputTextFlags_Password;
   182→        ImGui::SetCursorPos(pos);
   183→        if (t == "InputText")
   184→            ImGui::InputText(WLabel(c, "Text", "Yazı").c_str(), &buf[0], (int)buf.size(), fl);
   185→        else {
   186→            ImVec2 sz((float)(c->w ? c->w : atoi(c->Get("Width", "280").c_str())), (float)(c->h ? c->h : atoi(c->Get("Height", "120").c_str())));
   187→            ImGui::InputTextMultiline(WLabel(c, "Text", "Not").c_str(), &buf[0], buf.size(), sz, fl);
   188→        }
   189→        PushRect(c);
   190→    } else if (t == "InputInt" || t == "InputFloat") {
   191→        int iv = atoi(c->Get("Value", "0").c_str());
   192→        float fv = (float)atof(c->Get("Value", "0").c_str());
   193→        ImGui::SetCursorPos(pos);
   194→        if (t == "InputInt") ImGui::InputInt(WLabel(c, "Text", "Sayı").c_str(), &iv, atoi(c->Get("Step", "1").c_str()));
   195→        else ImGui::InputFloat(WLabel(c, "Text", "Sayı").c_str(), &fv, (float)atof(c->Get("Step", "1").c_str()));
   196→        PushRect(c);
   197→    } else if (t == "Combo" || t == "ListBox") {
   198→        int sel = atoi(c->Get("Selected", "0").c_str());

   200→        static std::vector<std::string> s_store;      // kare içi ömür güvenli depo
   201→        s_store.clear();
   202→        for (auto& i : items) s_store.push_back(WideStrToUtf8(i));
   203→        std::vector<const char*> ptrs;
   204→        for (auto& s : s_store) ptrs.push_back(s.c_str());
   205→        const char* cur = (sel >= 0 && sel < (int)ptrs.size()) ? ptrs[sel] : "";
   206→        ImGui::SetCursorPos(pos);
   207→        if (t == "Combo") {
   208→            if (ImGui::BeginCombo(WLabel(c, "Text", "Seçim").c_str(), cur)) ImGui::EndCombo();
   209→            PushRect(c);
   210→        } else {
   211→            int hi = atoi(c->Get("HeightItems", "5").c_str());

   213→            if (ImGui::BeginListBox(WLabel(c, "Text", "Liste").c_str(), ImVec2(0.0f, hgt))) {
   214→                for (auto& p : ptrs) ImGui::Selectable((std::string(p) + "##" + c->name).c_str(), false);
   215→                ImGui::EndListBox();
   216→            }
   217→            PushRect(c);
   218→        }
   219→    } else if (t == "ColorEdit4") {
   220→        ImVec4 col(atof(c->Get("R", "255").c_str()) / 255.0f,
   221→                    atof(c->Get("G", "255").c_str()) / 255.0f,

   228→    } else if (t == "ProgressBar") {
   229→        ImGui::SetCursorPos(pos);
   230→        ImVec2 sz((float)(c->w ? c->w : atoi(c->Get("Width", "200").c_str())), (float)(c->h ? c->h : atoi(c->Get("Height", "18").c_str())));
   231→        const char* ov = c->Get("Overlay", "").c_str();

   251→                        atof(c->Get("G", "200").c_str()) / 255.0f,

   348→    objstyle::Pop(stylePacked);

   352→//  tıklama noktasındaki üstteki nesne (tray dahil değil)
   353→static WidgetRect* HitUnder(ImVec2 pt) {
   354→    for (int i = (int)s_hitRects.size() - 1; i >= 0; i--) {
   355→        WidgetRect& r = s_hitRects[i];
   356→        if (r.tray) continue;
   357→        if (pt.x >= r.rmin.x && pt.x <= r.rmax.x && pt.y >= r.rmin.y && pt.y <= r.rmax.y) return &r;
   358→    }
   359→    return nullptr;
   360→}
   361→void ClickSelectUnder(ImVec2 pt, bool additive) {
   362→    WidgetRect* r = HitUnder(pt);
   363→    if (r) Select(r->c, additive);
   364→    else SelectForm();

   380→// 0:SO 1:O 2:DO 3:D 4:DD 5:G 6:SD 7:S
   381→static void HandleCorners(ImVec2 rmin, ImVec2 rmax, ImVec2 out[8]) {
   382→    out[0] = rmin;
   383→    out[1] = ImVec2((rmin.x + rmax.x) / 2, rmin.y);
   384→    out[2] = ImVec2(rmax.x, rmin.y);
   385→    out[3] = ImVec2(rmax.x, (rmin.y + rmax.y) / 2);
   386→    out[4] = rmax;
   387→    out[5] = ImVec2((rmin.x + rmax.x) / 2, rmax.y);
   388→    out[6] = ImVec2(rmin.x, rmax.y);
   389→    out[7] = ImVec2(rmin.x, (rmin.y + rmax.y) / 2);
   390→}
   391→static int HandleIndexAt(ImVec2 pt, ImVec2 rmin, ImVec2 rmax, float hs) {
   392→    ImVec2 c[8];
   393→    HandleCorners(rmin, rmax, c);
   394→    for (int i = 0; i < 8; i++)
   395→        if (fabsf(pt.x - c[i].x) <= hs && fabsf(pt.y - c[i].y) <= hs) return i;
   396→    return -1;
   397→}

   408→// sürükleme sırasında uygulanacak tutamaç (nesne form-göreli)
   409→static void ApplyHandle(Ctrl* c, int handle, ImVec2 d) {
   410→    int nx = c->x, ny = c->y, nw = s_origW, nh = s_origH;
   411→    if (handle == 0 || handle == 6 || handle == 7) { nw = s_origW - (int)d.x; nx = s_origX + (int)d.x; }
   412→    if (handle == 0 || handle == 1 || handle == 2) { nh = s_origH - (int)d.y; ny = s_origY + (int)d.y; }
   413→    if (handle == 2 || handle == 3 || handle == 4) nw = s_origW + (int)d.x;
   414→    if (handle == 4 || handle == 5 || handle == 6) nh = s_origH + (int)d.y;
   415→    nw = ImMax(nw, 12); nh = ImMax(nh, 8);
   416→    if (g_snapEnabled) {
   417→        nx = (nx + GRID_SIZE / 2) / GRID_SIZE * GRID_SIZE;
   418→        ny = (ny + GRID_SIZE / 2) / GRID_SIZE * GRID_SIZE;
   419→        nw = (nw + GRID_SIZE / 2) / GRID_SIZE * GRID_SIZE;
   420→        nh = (nh + GRID_SIZE / 2) / GRID_SIZE * GRID_SIZE;
   421→    }
   422→    c->x = nx; c->y = ny; c->w = nw; c->h = nh;
   423→    g_modelDirty = true;
   424→}
   425→static void ApplyFormHandle(Model& m, int handle, ImVec2 d) {
   426→    int nw = m.formW, nh = m.formH;
   427→    if (handle == 2 || handle == 3 || handle == 4) nw += (int)d.x;
   428→    if (handle == 4 || handle == 5 || handle == 6) nh += (int)d.y;
   429→    nw = ImMax(nw, 120); nh = ImMax(nh, 80);
   430→    if (g_snapEnabled) {
   431→        nw = (nw + GRID_SIZE / 2) / GRID_SIZE * GRID_SIZE;
   432→        nh = (nh + GRID_SIZE / 2) / GRID_SIZE * GRID_SIZE;
   433→    }
   434→    m.formW = nw; m.formH = nh;
   435→    m.formProps["ClientSize"] = std::to_string(m.formW) + ", " + std::to_string(m.formH);
   436→    g_modelDirty = true;
   437→}

   749→    // ================= 5) seçim vurguları (foreground) =======================
   750→    ImDrawList* fg = ImGui::GetForegroundDrawList();
   751→    // ikincil seçimler: in çerçeve
   752→    for (Ctrl* c : g_model.ValidSelection()) {
   753→        if (c == primary) continue;
   754→        for (auto& r : s_hitRects)
   755→            if (r.c == c && !r.tray) { DrawSelFrame(fg, r.rmin, r.rmax, false, false); break; }

   757→    // birincil seçim: dolu tutamaçlı çerçeve
   758→    if (havePrimaryRect) DrawSelFrame(fg, prMin, prMax, true, false);
   759→    // form seçimi: form çerçevesi + köşe tutamaçları
   760→    if (m.formSelected)
   761→        DrawSelFrame(fg, s_formPos, ImVec2(s_formPos.x + s_formSize.x, s_formPos.y + s_formSize.y), true, true);
   762→
   763→    // ================= 6) yerleştirme modu göstergesi ========================
   764→    if (!g_armTool.empty()) {
   765→        fg->AddRect(rectMin, rectMax, IM_COL32(90, 200, 120, 200), 4.0f, 0, 2.0f);
   766→        char b[160];
   767→        snprintf(b, sizeof b, "YERLE\305\236T\304\260R: %s  (form'a t\304\261kla — ESC iptal)", g_armTool.c_str());

   769→        fg->AddRectFilled(ImVec2(rectMin.x + 8, rectMax.y - ts.y - 12), ImVec2(rectMin.x + 16 + ts.x, rectMax.y - 8), IM_COL32(20, 60, 30, 220), 3);
   770→        fg->AddText(ImVec2(rectMin.x + 12, rectMax.y - ts.y - 8), IM_COL32(120, 255, 150, 255), b);
   771→    }
   772→
   773→    // rubber band çizgisi (kesikli dikdörtgen)
   774→    if (s_band) {
   775→        ImVec2 bmin(ImMin(s_bandStart.x, s_bandCur.x), ImMin(s_bandStart.y, s_bandCur.y));
   776→        ImVec2 bmax(ImMax(s_bandStart.x, s_bandCur.x), ImMax(s_bandStart.y, s_bandCur.y));
   777→        for (float x = bmin.x; x < bmax.x; x += 6.0f) {
   778→            fg->AddLine(ImVec2(x, bmin.y), ImVec2(ImMin(x + 3, bmax.x), bmin.y), IM_COL32(255, 176, 32, 220), 1.0f);

   781→        for (float y = bmin.y; y < bmax.y; y += 6.0f) {

   786→        snprintf(b, sizeof b, "%d x %d", (int)(bmax.x - bmin.x), (int)(bmax.y - bmin.y));
   787→        fg->AddText(ImVec2(bmin.x, bmin.y - 16), IM_COL32(255, 200, 120, 255), b);
   788→    }
   789→
   790→    // canvas çerçevesi
   791→    fg->AddRect(rectMin, rectMax, IM_COL32(20, 22, 28, 255), 0.0f, 0, 1.0f);
   792→    (void)haveFormRect;
   793→}
