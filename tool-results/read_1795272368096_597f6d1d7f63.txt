    41→// ayraç sürükleme yardımcı (yayın çizgisiyle)
    42→static void Splitter(const char* id, float* size, float min, float max, bool verticalAxis) {
    43→    ImGui::PushID(id);
    44→    ImVec2 p = ImGui::GetCursorScreenPos();
    45→    ImVec2 sz = verticalAxis ? ImVec2(4, ImGui::GetContentRegionAvail().y)
    46→                             : ImVec2(ImGui::GetContentRegionAvail().x, 4);
    47→    ImGui::InvisibleButton(id, sz, ImGuiButtonFlags_MouseButtonLeft);
    48→    if (ImGui::IsItemActive() && ImGui::IsMouseDragging(ImGuiMouseButton_Left)) {
    49→        float delta = verticalAxis ? ImGui::GetIO().MouseDelta.x : ImGui::GetIO().MouseDelta.y;
    50→        *size += delta;
    51→        if (*size < min) *size = min;
    52→        if (*size > max) *size = max;
    53→    }
    54→    if (ImGui::IsItemHovered())
    55→        ImGui::SetMouseCursor(verticalAxis ? ImGuiMouseCursor_ResizeEW : ImGuiMouseCursor_ResizeNS);
    56→    ImDrawList* dl = ImGui::GetWindowDrawList();
    57→    ImU32 col = ImGui::GetColorU32(ImGui::IsItemActive() || ImGui::IsItemHovered()
    58→                                   ? ImGuiCol_SeparatorActive : ImGuiCol_Separator);
    59→    dl->AddRectFilled(p, ImVec2(p.x + sz.x, p.y + sz.y), col, 2.0f);
    60→    ImGui::PopID();
    61→}

    66→static bool OpenFileDialog(std::wstring& out, const wchar_t* filter, const wchar_t* defExt) {
    67→    wchar_t file[MAX_PATH] = L"";
    68→    OPENFILENAMEW ofn = { sizeof(ofn) };
    69→    ofn.hwndOwner = g_hMain;
    70→    ofn.lpstrFile = file; ofn.nMaxFile = MAX_PATH;
    71→    ofn.lpstrFilter = filter;
    72→    ofn.lpstrDefExt = defExt;
    73→    ofn.Flags = OFN_FILEMUSTEXIST | OFN_EXPLORER;
    74→    if (GetOpenFileNameW(&ofn)) { out = file; return true; }
    75→    return false;
    76→}
    77→static bool SaveFileDialog(std::wstring& out, const wchar_t* filter, const wchar_t* defExt) {
    78→    wchar_t file[MAX_PATH] = L"";
    79→    OPENFILENAMEW ofn = { sizeof(ofn) };
    80→    ofn.hwndOwner = g_hMain;
    81→    ofn.lpstrFile = file; ofn.nMaxFile = MAX_PATH;
    82→    ofn.lpstrFilter = filter;
    83→    ofn.lpstrDefExt = defExt;
    84→    ofn.Flags = OFN_OVERWRITEPROMPT | OFN_EXPLORER;
    85→    if (GetSaveFileNameW(&ofn)) { out = file; return true; }
    86→    return false;
    87→}
    88→static bool BrowseFolder(std::wstring& out, const wchar_t* title) {
    89→    BROWSEINFOW bi = {};
    90→    bi.hwndOwner = g_hMain;
    91→    bi.lpszTitle = title;
    92→    bi.ulFlags = BIF_RETURNONLYFSDIRS | BIF_NEWDIALOGSTYLE;
    93→    PIDLIST_ABSOLUTE pidl = SHBrowseForFolderW(&bi);
    94→    if (!pidl) return false;
    95→    wchar_t path[MAX_PATH];
    96→    if (!SHGetPathFromIDListW(pidl, path)) { CoTaskMemFree(pidl); return false; }
    97→    CoTaskMemFree(pidl);
    98→    out = path;
    99→    return true;
   100→}

   105→static void DoNew() {
   106→    if (g_modelDirty) {
   107→        // basit uyarı: kaydetmeden yeni tasarım
   108→    }
   109→    forms::InitNew();
   110→    s_projectPath.clear();
   111→    g_modelDirty = false;
   112→    Status(L"Yeni tasarım: Form1");
   113→}
   114→static void DoOpen(const std::wstring& path = L"") {
   115→    std::wstring p = path;
   116→    if (p.empty())
   117→        if (!OpenFileDialog(p, L"M2God Tasarım (*.m2god)\0*.m2god\0", L"m2god")) return;
   118→    if (forms::LoadProject(p)) {
   119→        s_projectPath = p;
   120→        g_modelDirty = false;
   121→        style::Apply(forms::ThemeId());
   122→        Status(L"Açıldı: " + GetFileName(p));
   123→    } else Status(L"Açma hatası: " + GetFileName(p));
   124→}
   125→static void DoSave(bool as = false) {
   126→    std::wstring p = s_projectPath;
   127→    if (p.empty() || as) {
   128→        if (!SaveFileDialog(p, L"M2God Tasarım (*.m2god)\0*.m2god\0", L"m2god")) return;
   129→    }
   130→    if (forms::SaveProject(p)) {
   131→        s_projectPath = p;
   132→        g_modelDirty = false;
   133→        Status(L"Kaydedildi: " + GetFileName(p));
   134→    } else Status(L"Kaydetme hatası");
   135→}
   136→static void DoGenerate() {
   137→    std::wstring folder = s_genFolder;
   138→    if (!BrowseFolder(folder, L"Kod üretilecek klasör")) return;
   139→    s_genFolder = folder;
   140→    std::wstring proj = !folder.empty() ? GetFileName(folder) : L"M2GodApp";
   141→    if (proj.empty() || proj == folder) proj = L"M2GodApp";
   142→    if (codegen::EmitProject(folder, proj))
   143→        Status(L"Kod üretildi: " + folder + L"\\" + proj + L".sln");
   144→    else Status(L"Kod üretilemedi (SDK imgui/ klasörü exe yanında olmalı)");
   145→}

   150→static void Shortcuts() {
   151→    ImGuiIO& io = ImGui::GetIO();
   152→    if (io.WantTextInput) return;
   153→    if (ImGui::IsKeyPressed(ImGuiKey_N) && io.KeyCtrl) DoNew();
   154→    if (ImGui::IsKeyPressed(ImGuiKey_O) && io.KeyCtrl) DoOpen();
   155→    if (ImGui::IsKeyPressed(ImGuiKey_S) && io.KeyCtrl) DoSave();
   156→    if (ImGui::IsKeyPressed(ImGuiKey_F6)) DoGenerate();
   157→    if (ImGui::IsKeyPressed(ImGuiKey_F12)) s_previewOpen = !s_previewOpen;
   158→    if (ImGui::IsKeyPressed(ImGuiKey_Z) && io.KeyCtrl && g_model.CanUndo()) { g_model.Undo(); g_modelDirty = true; }
   159→    if (ImGui::IsKeyPressed(ImGuiKey_Y) && io.KeyCtrl && g_model.CanRedo()) { g_model.Redo(); g_modelDirty = true; }
   160→    if (ImGui::IsKeyPressed(ImGuiKey_C) && io.KeyCtrl) canvas::CopySel();
   161→    if (ImGui::IsKeyPressed(ImGuiKey_X) && io.KeyCtrl) canvas::CutSel();
   162→    if (ImGui::IsKeyPressed(ImGuiKey_V) && io.KeyCtrl) {
   163→        // son fare konumuna (form merkezine) yapıştır
   164→        ImVec2 m = io.MousePos;
   165→        canvas::PasteAt((int)m.x, (int)m.y);
   166→    }
   167→    if (ImGui::IsKeyPressed(ImGuiKey_Delete) && g_model.selected) {
   168→        g_model.PushUndo();
   169→        g_model.Delete(g_model.selected);
   170→        g_modelDirty = true;
   171→    }
   172→    if (ImGui::IsKeyPressed(ImGuiKey_T) && io.KeyCtrl && g_model.selected) { g_model.BringToFront(g_model.selected); g_modelDirty = true; }
   173→    if (ImGui::IsKeyPressed(ImGuiKey_B) && io.KeyCtrl && g_model.selected) { g_model.SendToBack(g_model.selected); g_modelDirty = true; }
   174→    if (ImGui::IsKeyPressed(ImGuiKey_F7) && vsint::IsConnected())
   175→        vsint::SyncToProject(g_model.selected, L"", nullptr);
   176→}

   338→            insts = vsint::Enumerate();

   340→        }
   341→        ImGui::Separator();
   342→        if (insts.empty()) {
   343→            ImGui::TextColored(ImVec4(1, 0.8f, 0.4f, 1),
   344→                "Calisan Visual Studio bulunamadi (devenv.exe).\nBaglanmak icin VS'yi bir .sln ile acin, sonra Yenile.");
   345→        } else {
   346→            ImGui::TextUnformatted("1) Calisan Visual Studio ornegini secin:");
   347→            static int sel = -1;
   348→            for (int i = 0; i < (int)insts.size(); i++) {
   349→                char lbl[256];
   350→                snprintf(lbl, sizeof(lbl), "%s  [%s]%s",
   351→                         M2God::W2U(insts[i].title).c_str(),

   359→                static std::vector<std::wstring> projects;

   363→                    projects = vsint::EnumerateProjectsOfSolution(insts[sel].solutionPath);

   371→                    const char* names[64];

   377→            }
   378→            ImGui::Separator();

   380→                std::wstring projPath;
   381→                if (!insts[sel].solutionPath.empty()) {

   383→                    if (projects.empty())
   384→                        projects = vsint::FindVcxprojsUnder(GetFileDir(insts[sel].solutionPath), 3);
   385→                    if (!projects.empty()) projPath = projects[0];
   386→                }
   387→                if (vsint::Connect(insts[sel], projPath)) {
   388→                    vsint::PrepareProjectFiles();
   389→                    Status(L"VS bağlandı: " + vsint::Current().projectName);
   390→                    ImGui::CloseCurrentPopup();
   391→                } else Status(L"Bağlanamadı: .vcxproj bulunamadı");
   392→            }
   393→            ImGui::SameLine();
   394→            if (ImGui::Button("Kapat", ImVec2(110, 0))) ImGui::CloseCurrentPopup();
   395→        }
   396→        ImGui::EndPopup();
   397→    }
   398→    // hakkında
   399→    if (ImGui::BeginPopupModal("Hakkında", nullptr, ImGuiWindowFlags_AlwaysAutoResize)) {
   400→        ImGui::TextUnformatted("M2God ImGui Editor — IMGUI_GOD_SSSTIER");
   401→        ImGui::BulletText("M2GodWin32 Faz2.13 Win32 UI editorunun ImGui portu");
   402→        ImGui::BulletText("UI editör KENDISI de ImGui (tek OS penceresi)");

   668→int RunCli(const std::wstring& cmdLine) {
   669→    std::wstring cl = cmdLine;
   670→    if (cl.find(L"--gensample") != std::wstring::npos) {
   671→        std::wstring dir;
   672→        size_t p = cl.find(L"--gensample");
   673→        p += wcslen(L"--gensample");
   674→        while (p < cl.size() && cl[p] == L' ') p++;
   675→        while (p < cl.size() && cl[p] != L' ') dir += cl[p++];
   676→        if (dir.empty() || dir[0] == L'-') dir = L"DenemImgui";
   677→        // örnek tasarım: Form1 + Form2 (ShowForm köprüsü görünsün)
   678→        forms::InitNew();
   679→        g_model.formProps[L"Text"] = L"Ana Form";
   680→        g_model.formW = 480; g_model.formH = 360;
   681→        g_model.formProps[L"ClientSize"] = L"480, 360";
   682→        g_model.formProps[L"BackColor"] = L"#23232D";
   683→        g_model.Add(L"Label", 24, 30)->props[L"Text"] = L"M2God ImGui Editor örnek projesi";
   684→        if (Ctrl* b = g_model.Add(L"Button", 24, 70)) {
   685→            b->props[L"Text"] = L"Form2'yi aç";
   686→            b->w = 130; b->h = 34;
   687→            b->props[L"Size"] = L"130, 34";
   688→            b->events[L"Click"] = L"button1_Click";
   689→        }
   690→        if (Ctrl* cb = g_model.Add(L"CheckBox", 24, 120)) {
   691→            cb->props[L"Text"] = L"Seçenek kutusu";
   692→            cb->events[L"CheckedChanged"] = L"checkBox1_CheckedChanged";
   693→        }
   694→        if (Ctrl* s = g_model.Add(L"SliderInt", 24, 160)) {
   695→            s->props[L"Min"] = L"0"; s->props[L"Max"] = L"100"; s->props[L"Value"] = L"42";
   696→            s->events[L"ValueChanged"] = L"sliderInt1_ValueChanged";
   697→        }
   698→        if (Ctrl* t = g_model.Add(L"Timer", 0, 0)) {
   699→            t->props[L"Interval"] = L"1000";
   700→            t->events[L"Tick"] = L"timer1_Tick";
   701→        }
   702→        g_model.formEvents[L"Load"] = L"Form1_Load";
   703→        // Form2
   704→        forms::AddNew();   // aktif Form2
   705→        g_model.formProps[L"Text"] = L"İkinci Form";
   706→        g_model.formW = 360; g_model.formH = 240;
   707→        g_model.formProps[L"ClientSize"] = L"360, 240";
   708→        if (Ctrl* l = g_model.Add(L"Label", 20, 24)) l->props[L"Text"] = L"Ben Form2";
   709→        if (Ctrl* b2 = g_model.Add(L"Button", 20, 60)) {
   710→            b2->props[L"Text"] = L"Kapat";
   711→            b2->events[L"Click"] = L"button1_Click";
   712→        }
   713→        forms::SwitchTo(0);
   714→        // handler gövdelerine örnek kod yaz (senkron korumalı akışta korunur):
   715→        // (üretim + elle dokunma simülasyonu — dosyayı ürettikten sonra gövdeyi değiştir)
   716→        if (codegen::EmitProject(dir, L"DenemImgui")) {
   717→                std::wstring cppPath = dir + L"\\DenemImgui\\Form1.cpp";
   718→            std::string data;
   719→            if (ReadFileText(cppPath, data)) {
   720→                std::wstring cpp = Utf8ToWide(data);
   721→                // button1_Click gövdesine ShowForm yaz
   722→                std::wstring anchor = L"void button1_Click() {\n    // TODO: button1_Click";
   723→                if (cpp.find(anchor) != std::wstring::npos)
   724→                    cpp = ReplaceAll(cpp, anchor,
   725→                        std::wstring(L"void button1_Click() {\n") +
   726→                        L"    M2God::ShowForm(L\"Form2\");   // ikinci formu açar");
   727→                WriteFileText(cppPath, WideToUtf8(cpp));
   728→            }
   729→            return 99;   // tamam → çık
   730→        }
   731→        return 98;      // hata
   732→    }
   733→    if (cl.find(L"--open") != std::wstring::npos) {
   734→        size_t p = cl.find(L"--open") + wcslen(L"--open");
   735→        while (p < cl.size() && cl[p] == L' ') p++;
   736→        std::wstring file;
   737→        while (p < cl.size() && cl[p] != L' ') file += cl[p++];
   738→        if (!file.empty() && FileExists(file)) {
   739→            forms::LoadProject(file);
   740→            s_projectPath = file;
   741→        }
   742→        return 0;   // UI ile devam
   743→    }
   744→    return 0;
   745→}
