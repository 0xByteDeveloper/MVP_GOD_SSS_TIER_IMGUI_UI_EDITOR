    59→static void BuildPreview(Preview& pv) {
    60→    // eski ağacı temizle (Roots silinir → çocuklar Widget yıkıcısıyla özyinelemeli)
    61→    for (auto* w : pv.objs.Roots) delete w;
    62→    pv.objs = M2God::FormObjects();
    63→    pv.byCtrl.clear();
    64→    pv.ctrlOf.clear();
    65→    pv.ok = true;
    66→    // 1) tüm widget'ları yarat
    67→    for (auto* c : g_model.ctrls) {
    68→        if (_wcsicmp(c->type.c_str(), L"TabPage") == 0) {
    69→            // sayfalar WTabBar::Kids'a girer — kendi başına kökte yer almaz
    70→            M2God::WTabPage* pg = new M2God::WTabPage();
    71→            pv.byCtrl[c] = pg;
    72→            pv.ctrlOf[pg] = c;
    73→            continue;
    74→        }
    75→        M2God::Widget* w = M2God::Create(c->type.c_str());
    76→        if (!w) w = new M2God::WLabel();      // bilinmeyen tip → etiket (asla çökmez)
    77→        pv.byCtrl[c] = w;
    78→        pv.ctrlOf[w] = c;
    79→    }
    80→    // 2) özellikleri uygula (SDK sözleşmesi: editör + üretilen kod aynı yol)
    81→    for (auto& kv : pv.byCtrl) {
    82→        Ctrl* c = kv.first;
    83→        M2God::Widget* w = kv.second;
    84→        w->Name = c->name;
    85→        w->Pos  = ImVec2((float)c->x, (float)c->y);
    86→        w->Size = ImVec2((float)c->w, (float)c->h);
    87→        for (auto& p : c->props) w->Set(p.first.c_str(), p.second.c_str());
    88→    }
    89→    // 3) ebeveyn bağla (model sırası korunur → z-sırası)
    90→    for (auto* c : g_model.ctrls) {
    91→        M2God::Widget* w = pv.byCtrl[c];
    92→        M2God::Widget* pw = c->parent ? pv.byCtrl[c->parent] : nullptr;
    93→        pv.objs.Add(w, pw);
    94→    }
    95→}

   100→static ImVec2 ContainerContentOrigin(Ctrl* cont, const ImVec2& contAbs) {
   101→    std::wstring t = Lower(cont->type);
   102→    if (t == L"groupbox")         return ImVec2(contAbs.x + 10, contAbs.y + 10);
   103→    if (t == L"childwindow")      return ImVec2(contAbs.x + 8,  contAbs.y + 8);
   104→    if (t == L"tabpage")          return ImVec2(contAbs.x, contAbs.y);
   105→    if (t == L"collapsingheader") return ImVec2(contAbs.x, contAbs.y + 22);
   106→    return contAbs;
   107→}

   112→static Ctrl* Hit(int fx, int fy) {
   113→    for (int i = (int)g_model.ctrls.size() - 1; i >= 0; i--) {
   114→        Ctrl* c = g_model.ctrls[i];
   115→        if (c->nonVisual) continue;
   116→        // kapanık CollapsingHeader içindekiler görünmez
   117→        if (c->parent && _wcsicmp(c->parent->type.c_str(), L"CollapsingHeader") == 0) {
   118→            // varsayılan açık varsayımı: DefaultOpen != False ise görünür
   119→            if (_wcsicmp(c->parent->Get(L"DefaultOpen", L"True").c_str(), L"False") == 0) continue;
   120→        }
   121→        // pasif sayfa içi (TabBar'ın seçili olmayan sayfası) görünmez
   122→        if (c->parent && _wcsicmp(c->parent->type.c_str(), L"TabPage") == 0) {
   123→            Ctrl* tab = c->parent->parent;
   124→            if (tab) {
   125→                auto pages = TabPagesOf(tab);
   126→                int sel = s_pageSel.count(tab) ? s_pageSel[tab] : 0;
   127→                bool active = false;
   128→                for (int k = 0; k < (int)pages.size(); k++)
   129→                    if (pages[k] == c->parent && k == sel) active = true;
   130→                if (!active) continue;
   131→            }
   132→        }
   133→        POINT ap = g_model.AbsPos(c);
   134→        if (fx >= ap.x && fx <= ap.x + c->w && fy >= ap.y && fy <= ap.y + c->h)
   135→            return c;
   136→    }
   137→    return nullptr;
   138→}

   140→// bırakma hedefi konteyner: en derin, kendisi/yakını DEĞIL, içinde nokta
   141→static Ctrl* DropContainer(int fx, int fy, Ctrl* exclude) {
   142→    Ctrl* best = nullptr;
   143→    double bestArea = 1e18;
   144→    for (auto* c : g_model.ctrls) {
   145→        if (c == exclude || c->nonVisual) continue;
   146→        if (!IsContainerType(c->type)) continue;
   147→        if (exclude && g_model.IsAncestorOf(exclude, c)) continue;
   148→        if (_wcsicmp(c->type.c_str(), L"TabPage") == 0) continue;   // sayfa TabBar üstünden çözülür
   149→        POINT ap = g_model.AbsPos(c);
   150→        if (fx >= ap.x && fx <= ap.x + c->w && fy >= ap.y && fy <= ap.y + c->h) {
   151→            double area = (double)c->w * c->h;
   152→            if (area < bestArea) { bestArea = area; best = c; }
   153→        }
   154→    }
   155→    if (best && _wcsicmp(best->type.c_str(), L"TabBar") == 0) {
   156→        // sekme alanına bırakıldı → AKTİF sayfaya ebeveynlen
   157→        auto pages = TabPagesOf(best);
   158→        int sel = s_pageSel.count(best) ? s_pageSel[best] : 0;
   159→        if (sel < (int)pages.size()) best = pages[sel];
   160→    }
   161→    return best;
   162→}

   174→static void HandleRects(const Rect& r, Rect out[8]) {
   175→    // TL, TR, BL, BR, L, R, T, B
   176→    out[0] = { r.x0 - HANDLE, r.y0 - HANDLE, r.x0 + HANDLE, r.y0 + HANDLE };
   177→    out[1] = { r.x1 - HANDLE, r.y0 - HANDLE, r.x1 + HANDLE, r.y0 + HANDLE };
   178→    out[2] = { r.x0 - HANDLE, r.y1 - HANDLE, r.x0 + HANDLE, r.y1 + HANDLE };
   179→    out[3] = { r.x1 - HANDLE, r.y1 - HANDLE, r.x1 + HANDLE, r.y1 + HANDLE };
   180→    out[4] = { r.x0 - HANDLE, r.y0 + (r.y1 - r.y0) / 2 - HANDLE, r.x0 + HANDLE, r.y0 + (r.y1 - r.y0) / 2 + HANDLE };
   181→    out[5] = { r.x1 - HANDLE, r.y0 + (r.y1 - r.y0) / 2 - HANDLE, r.x1 + HANDLE, r.y0 + (r.y1 - r.y0) / 2 + HANDLE };
   182→    out[6] = { r.x0 + (r.x1 - r.x0) / 2 - HANDLE, r.y0 - HANDLE, r.x0 + (r.x1 - r.x0) / 2 + HANDLE, r.y0 + HANDLE };
   183→    out[7] = { r.x0 + (r.x1 - r.x0) / 2 - HANDLE, r.y1 - HANDLE, r.x0 + (r.x1 - r.x0) / 2 + HANDLE, r.y1 + HANDLE };
   184→}
   185→static bool InRect(const Rect& r, const ImVec2& p) {
   186→    return p.x >= r.x0 && p.x <= r.x1 && p.y >= r.y0 && p.y <= r.y1;
   187→}
   188→static void DrawHandles(ImDrawList* dl, const Rect& r, ImU32 accent) {
   189→    static const ImU32 white = IM_COL32(255, 255, 255, 255);
   190→    static const ImU32 black = IM_COL32(40, 40, 40, 255);
   191→    dl->AddRectFilled(ImVec2(r.x0, r.y0), ImVec2(r.x1, r.y1), accent & IM_COL32_A_MASK, 0.0f);   // dolgu
   192→    dl->AddRect(ImVec2(r.x0, r.y0), ImVec2(r.x1, r.y1), white, 0.0f, 0, 1.0f);
   193→    Rect hr[8];
   194→    HandleRects(r, hr);
   195→    for (auto& h : hr) {
   196→        dl->AddRectFilled(ImVec2(h.x0, h.y0), ImVec2(h.x1, h.y1), white);
   197→        dl->AddRect(ImVec2(h.x0, h.y0), ImVec2(h.x1, h.y1), black, 0.0f, 0, 1.0f);
   198→    }
   199→}

   204→static json::Value CtrlToJson(const Ctrl* c) {
   205→    json::Value o = json::Value::MakeObj();
   206→    o.Set(L"type", c->type);
   207→    o.Set(L"x", (double)c->x); o.Set(L"y", (double)c->y);
   208→    o.Set(L"w", (double)c->w); o.Set(L"h", (double)c->h);
   209→    json::Value p = json::Value::MakeObj();
   210→    for (auto& kv : c->props) p.Set(kv.first, kv.second);
   211→    o.Set(L"props", p);
   212→    json::Value ev = json::Value::MakeObj();
   213→    for (auto& kv : c->events) ev.Set(kv.first, kv.second);
   214→    o.Set(L"events", ev);
   215→    json::Value kids = json::Value::MakeArr();
   216→    for (auto* k : g_model.ChildrenOf(const_cast<Ctrl*>(c)))
   217→        kids.Push(CtrlToJson(k));
   218→    o.Set(L"children", kids);
   219→    return o;
   220→}

   238→static Ctrl* PasteTree(const json::Value& o, Ctrl* parent, int dx, int dy) {
   239→    std::wstring type = o.Get(L"type").AsStr();
   240→    Ctrl* c = g_model.Add(type, 0, 0, parent, false);
   241→    if (!c) return nullptr;
   242→    c->x = (int)o.Get(L"x").AsNum() + dx;
   243→    c->y = (int)o.Get(L"y").AsNum() + dy;
   244→    c->w = (int)o.Get(L"w").AsNum();
   245→    c->h = (int)o.Get(L"h").AsNum();
   246→    const json::Value& p = o.Get(L"props");
   247→    if (p.IsObj()) for (auto& kv : p.obj) c->props[kv.first] = kv.second.AsStr();
   248→    wchar_t b[64]; swprintf(b, 64, L"%d, %d", c->x, c->y); c->props[L"Location"] = b;
   249→    swprintf(b, 64, L"%d, %d", c->w, c->h); c->props[L"Size"] = b;
   250→    const json::Value& ev = o.Get(L"events");
   251→    if (ev.IsObj()) for (auto& kv : ev.obj) c->events[kv.first] = kv.second.AsStr();
   252→    // çocuksa ebeveyn içi geometri; özelse kökte konum
   253→    const json::Value& kids = o.Get(L"children");
   254→    if (kids.IsArr())
   255→        for (auto& k : kids.arr) PasteTree(k, c, 0, 0);
   256→    return c;
   257→}
   258→bool PasteAt(int fx, int fy) {
   259→    if (g_clip.empty()) return false;
   260→    bool ok = false;
   261→    json::Value v = json::Parse(g_clip, &ok);
   262→    if (!ok) return false;
   263→    g_model.PushUndo();
   264→    Ctrl* parent = DropContainer(fx, fy, nullptr);
   265→    ImVec2 origin(0, 0);
   266→    int px = fx, py = fy;
   267→    if (parent) {
   268→        POINT ap = g_model.AbsPos(parent);
   269→        ImVec2 co = ContainerContentOrigin(parent, ImVec2((float)ap.x, (float)ap.y));
   270→        px = (int)(fx - co.x);
   271→        py = (int)(fy - co.y);
   272→    }
   273→    Ctrl* c = PasteTree(v, parent, px, py);
   274→    if (c) {
   275→        g_model.selected = c;
   276→        g_model.formSelected = false;
   277→        Changed();
   278→        Status(L"Yapıştırıldı: " + c->name);
   279→        return true;
   280→    }
   281→    return false;
   282→}

   294→static const int TAB_STRIP_H = 24;
   295→
   296→static void DrawTabStrip(ImDrawList* dl, Ctrl* tab, const ImVec2& abs, Preview& pv,
   297→                         ImVec2& pageOrigin, ImVec2& pageSize) {
   298→    auto pages = TabPagesOf(tab);
   299→    int sel = s_pageSel.count(tab) ? s_pageSel[tab] : 0;
   300→    if (sel >= (int)pages.size()) sel = (int)pages.size() - 1;
   301→    if (sel < 0) sel = 0;
   302→    s_pageSel[tab] = sel;
   303→    ImU32 bgDim   = ImGui::GetColorU32(ImGuiCol_TitleBg);
   304→    ImU32 bgActive = ImGui::GetColorU32(ImGuiCol_TitleBgActive);
   305→    ImU32 bgIdle  = ImGui::GetColorU32(ImGuiCol_TabUnfocused, 0.85f);
   306→    ImU32 bgHover = ImGui::GetColorU32(ImGuiCol_TabUnfocusedActive, 0.9f);
   307→    ImU32 textCol = ImGui::GetColorU32(ImGuiCol_Text);
   308→    ImVec2 stripMax(abs.x + tab->w, abs.y + TAB_STRIP_H);
   309→    dl->AddRectFilled(abs, stripMax, bgDim);
   310→    dl->AddRect(abs, ImVec2(abs.x + tab->w, abs.y + tab->h), ImGui::GetColorU32(ImGuiCol_Border), 4.0f);
   311→    float tx = abs.x + 4;
   312→    for (int i = 0; i < (int)pages.size(); i++) {
   313→        std::wstring t = pages[i]->Get(L"Text");
   314→        if (t.empty()) t = pages[i]->name;
   315→        ImVec2 ts = ImGui::CalcTextSize(M2God::W2U(t).c_str());
   316→        float w = ts.x + 18.0f;
   317→        ImVec2 a(tx, abs.y + 3), b(tx + w, abs.y + TAB_STRIP_H);
   318→        dl->AddRectFilled(a, b, i == sel ? bgActive : bgIdle, 4.0f, ImDrawFlags_RoundCornersTop);
   319→        dl->AddText(ImVec2(a.x + 9, a.y + (TAB_STRIP_H - 3 - ts.y) * 0.5f), textCol, M2God::W2U(t).c_str());
   320→        // sekme tıklaması (tasarım etkileşimi — form kilidiyle)
   321→        ImVec2 mp = ImGui::GetIO().MousePos;
   322→        if (mp.x >= a.x && mp.x <= b.x && mp.y >= a.y && mp.y <= b.y) {
   323→            if (i != sel) dl->AddRect(a, b, bgHover, 4.0f, ImDrawFlags_RoundCornersTop, 1.5f);
   324→            if (ImGui::IsMouseClicked(ImGuiMouseButton_Left))
   325→                s_pageSel[tab] = i;
   326→        }
   327→        tx += w;
   328→    }
   329→    pageOrigin = ImVec2(abs.x + 4, abs.y + TAB_STRIP_H + 4);
   330→    pageSize   = ImVec2((float)tab->w - 8, (float)tab->h - TAB_STRIP_H - 8);
   331→}

   336→static void DrawComponentTray(ImDrawList* dl, const ImVec2& origin, const ImVec2& size) {
   337→    dl->AddRectFilled(origin, ImVec2(origin.x + size.x, origin.y + size.y),
   338→                      ImGui::GetColorU32(ImGuiCol_TabUnfocused), 4.0f);
   339→    float x = origin.x + 8;
   340→    float cy = origin.y + size.y * 0.5f;
   341→    dl->AddText(ImVec2(x, cy - 26), ImGui::GetColorU32(ImGuiCol_TextDisabled), "Bilesenler");
   342→    x += 90;
   343→    for (auto* c : g_model.ctrls) {
   344→        if (!c->nonVisual) continue;
   345→        std::string nm = M2God::W2U(c->name + L" (" + c->type + L")");
   346→        ImVec2 ts = ImGui::CalcTextSize(nm.c_str());
   347→        ImVec2 a(x, cy - ts.y * 0.5f - 3), b(x + ts.x + 16, cy + ts.y * 0.5f + 3);
   348→        bool sel = g_model.selected == c;
   349→        dl->AddRectFilled(a, b, ImGui::GetColorU32(ImGuiCol_FrameBg, 0.85f), 3.0f);
   350→        dl->AddRect(a, b, sel ? ImGui::GetColorU32(ImGuiCol_ButtonActive)
   351→                              : ImGui::GetColorU32(ImGuiCol_Border), 3.0f, 0, 1.0f);
   352→        dl->AddText(ImVec2(a.x + 8, a.y + 3), ImGui::GetColorU32(ImGuiCol_Text), nm.c_str());
   353→        ImVec2 mp = ImGui::GetIO().MousePos;
   354→        if (mp.x >= a.x && mp.x <= b.x && mp.y >= a.y && mp.y <= b.y) {
   355→            if (ImGui::IsMouseClicked(ImGuiMouseButton_Left)) {
   356→                g_model.selected = c;
   357→                g_model.formSelected = false;
   358→            }
   359→        }
   360→        x += ts.x + 26;
   361→        if (x > origin.x + size.x - 60) break;
   362→    }
   363→}

   369→static void RenderPreviewTree(Preview& pv, Ctrl* tabRoot, const ImVec2& origin) {
   370→    // 1) görsel olmayanlar yok; TabBar özel çizilir
   371→    for (auto* w : pv.objs.Roots) {
   372→        Ctrl* c = pv.ctrlOf.count(w) ? pv.ctrlOf[w] : nullptr;
   373→        if (!c) continue;
   374→        if (c->nonVisual) continue;
   375→        if (_wcsicmp(c->type.c_str(), L"TabBar") == 0) {
   376→            ImVec2 abs(origin.x + c->x, origin.y + c->y);
   377→            ImDrawList* dl = ImGui::GetWindowDrawList();
   378→            ImVec2 pageOrigin, pageSize;
   379→            // önce arka plan + sayfa İÇERİĞİ (çocuklar SDK Render'ıyla)
   380→            auto pages = TabPagesOf(c);
   381→            int sel = s_pageSel.count(c) ? s_pageSel[c] : 0;
   382→            if (sel >= (int)pages.size()) sel = (int)pages.size() - 1;
   383→            // içerik: seçili sayfanın çocukları (SDK yol — WTabPage::RenderKids)
   384→            if (sel >= 0 && sel < (int)pages.size()) {
   385→                auto itW = pv.byCtrl.find(pages[sel]);
   386→                if (itW != pv.byCtrl.end()) {
   387→                    M2God::WTabPage* pg = (M2God::WTabPage*)itW->second;
   388→                    ImVec2 pgOrigin(abs.x + 4, abs.y + TAB_STRIP_H + 4);
   389→                    // çerçeve
   390→                    dl->AddRectFilled(ImVec2(abs.x + 2, abs.y + TAB_STRIP_H + 2),
   391→                                      ImVec2(abs.x + c->w - 2, abs.y + c->h - 2),
   392→                                      ImGui::GetColorU32(ImGuiCol_WindowBg, 0.4f));
   393→                    pg->RenderKids(pgOrigin);
   394→                }
   395→            }

   398→            DrawTabStrip(dl, c, abs, pv, po, ps);
   399→            continue;
   400→        }
   401→        w->Render(origin);
   402→    }
   403→}

   666→// sürükleme durumu yönetimi
   667→// ---------------------------------------------------------------------------
   668→static void BeginDrag(DragMode m, Ctrl* c) {
   669→    s_drag.mode = m;
   670→    s_drag.ctrl = c;
   671→    s_drag.moved = false;
   672→    s_drag.undoPushed = false;
   673→    if (c) {
   674→        s_drag.sx = c->x; s_drag.sy = c->y;
   675→        s_drag.sw = c->w; s_drag.sh = c->h;
   676→    } else {
   677→        s_drag.fsw = g_model.formW;
   678→        s_drag.fsh = g_model.formH;
   679→    }
   680→}
   681→static void Snap(int& v) { if (g_snap) v = ((v + GRID_SIZE / 2) / GRID_SIZE) * GRID_SIZE; }
   682→
   683→static void UpdateDrag(const ImVec2& mp, const ImVec2& paper) {
   684→    Ctrl* c = s_drag.ctrl;
   685→    if (!s_drag.undoPushed) {
   686→        if (s_drag.mode == DR_MOVE || s_drag.mode >= DR_SZ_TL) {
   687→            g_model.PushUndo();      // ilk gerçek hareket öncesi kayıt
   688→            s_drag.undoPushed = true;
   689→        }
   690→    }
   691→    s_drag.moved = true;
   692→    int fx = (int)(mp.x - paper.x), fy = (int)(mp.y - paper.y);
   693→    if (!c) {     // form boyutlandırma
   694→        int W = fx, H = fy;
   695→        switch (s_drag.mode) {
   696→        case DR_SZ_TL: case DR_SZ_TR: case DR_SZ_BL: case DR_SZ_BR: W = fx; H = fy; break;
   697→        case DR_SZ_L: W = fx; H = g_model.formH; break;
   698→        case DR_SZ_R: W = fx; H = g_model.formH; break;
   699→        case DR_SZ_T: W = g_model.formW; H = fy; break;
   700→        case DR_SZ_B: W = g_model.formW; H = fy; break;
   701→        default: return;
   702→        }
   703→        if (W < 60) W = 60;
   704→        if (H < 40) H = 40;
   705→        Snap(W); Snap(H);
   706→        g_model.formW = W;
   707→        g_model.formH = H;
   708→        wchar_t b[64]; swprintf(b, 64, L"%d, %d", W, H);
   709→        g_model.formProps[L"ClientSize"] = b;
   710→        g_modelDirty = true;
   711→        return;
   712→    }
   713→    // taşıma: mutlak konum → ebeveyn görelisi
   714→    if (s_drag.mode == DR_MOVE) {
   715→        int ax = (int)(mp.x - paper.x - s_drag.grabOff.x);
   716→        int ay = (int)(mp.y - paper.y - s_drag.grabOff.y);
   717→        if (c->parent) {
   718→            POINT ap = g_model.AbsPos(c->parent);
   719→            ImVec2 co = ContainerContentOrigin(c->parent, ImVec2((float)ap.x, (float)ap.y));
   720→            ax = (int)(ax - co.x);
   721→            ay = (int)(ay - co.y);
   722→            if (ax < -c->w + 6) ax = -c->w + 6;      // ebeveyn içinde en az 6px görünsün
   723→            if (ay < -c->h + 6) ay = -c->h + 6;
   724→        }
   725→        Snap(ax); Snap(ay);
   726→        if (ax != c->x || ay != c->y) {
   727→            c->x = ax; c->y = ay;
   728→            wchar_t b[64]; swprintf(b, 64, L"%d, %d", ax, ay);
   729→            c->props[L"Location"] = b;
   730→            g_modelDirty = true;
   731→        }
   732→        return;
   733→    }
   734→    if (s_drag.mode >= DR_SZ_TL) {
   735→        // genel boyutlandırma: başlangıç kutusu + delta (mutlak form koordinatında)
   736→        int x0 = s_drag.sx, y0 = s_drag.sy;         // ebeveyn görelisi başlangıç
   737→        POINT ap0 = g_model.AbsPos(c);
   738→        int X0 = ap0.x, Y0 = ap0.y;                // başlangıç mutlak (form) kutusu
   739→        int X1 = X0 + s_drag.sw, Y1 = Y0 + s_drag.sh;
   740→        switch (s_drag.mode) {
   741→        case DR_SZ_TL: X0 = fx; Y0 = fy; break;
   742→        case DR_SZ_TR: X1 = fx; Y0 = fy; break;
   743→        case DR_SZ_BL: X0 = fx; Y1 = fy; break;
   744→        case DR_SZ_BR: X1 = fx; Y1 = fy; break;
   745→        case DR_SZ_L:  X0 = fx; break;
   746→        case DR_SZ_R:  X1 = fx; break;
   747→        case DR_SZ_T:  Y0 = fy; break;
   748→        case DR_SZ_B:  Y1 = fy; break;
   749→        default: break;
   750→        }
   751→        (void)x0; (void)y0;
   752→        if (X1 - X0 < 4) X1 = X0 + 4;
   753→        if (Y1 - Y0 < 4) Y1 = Y0 + 4;
   754→        // ebeveyne geri çevir
   755→        int nx = X0, ny = Y0, nw = X1 - X0, nh = Y1 - Y0;
   756→        if (c->parent) {
   757→            POINT pap = g_model.AbsPos(c->parent);
   758→            ImVec2 co = ContainerContentOrigin(c->parent, ImVec2((float)pap.x, (float)pap.y));
   759→            nx = (int)(X0 - co.x);
   760→            ny = (int)(Y0 - co.y);
   761→        }
   762→        c->x = nx; c->y = ny; c->w = nw; c->h = nh;
   763→        wchar_t b[64];
   764→        swprintf(b, 64, L"%d, %d", nx, ny); c->props[L"Location"] = b;
   765→        swprintf(b, 64, L"%d, %d", nw, nh); c->props[L"Size"] = b;
   766→        g_modelDirty = true;
   767→    }
   768→}
   769→static void EndDrag() {
   770→    if (s_drag.moved) {
   771→        Changed();
   772→        if (s_drag.ctrl) Status(L"Taşındı: " + s_drag.ctrl->name);
   773→        else Status(L"ClientSize: " + std::to_wstring(g_model.formW) + L", " + std::to_wstring(g_model.formH));
   774→    }
   775→    s_drag.mode = DR_NONE;
   776→    s_drag.ctrl = nullptr;
   777→}
   778→static void CancelDrag() {
   779→    if (s_drag.ctrl) {
   780→        s_drag.ctrl->x = s_drag.sx; s_drag.ctrl->y = s_drag.sy;
   781→        s_drag.ctrl->w = s_drag.sw; s_drag.ctrl->h = s_drag.sh;
   782→        wchar_t b[64];
   783→        swprintf(b, 64, L"%d, %d", s_drag.ctrl->x, s_drag.ctrl->y);
   784→        s_drag.ctrl->props[L"Location"] = b;
   785→        swprintf(b, 64, L"%d, %d", s_drag.ctrl->w, s_drag.ctrl->h);
   786→        s_drag.ctrl->props[L"Size"] = b;
   787→    } else {
   788→        g_model.formW = s_drag.fsw; g_model.formH = s_drag.fsh;
   789→        wchar_t b[64]; swprintf(b, 64, L"%d, %d", g_model.formW, g_model.formH);
   790→        g_model.formProps[L"ClientSize"] = b;
   791→    }
   792→    s_drag.mode = DR_NONE;
   793→    s_drag.ctrl = nullptr;
   794→    Status(L"İptal edildi");
   795→}
