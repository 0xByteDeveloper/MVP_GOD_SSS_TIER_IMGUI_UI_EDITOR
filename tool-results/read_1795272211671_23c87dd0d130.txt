    59→    59→static void BuildPreview(Preview& pv) {
    60→    60→    // eski ağacı temizle (Roots silinir → çocuklar Widget yıkıcısıyla özyinelemeli)
    61→    61→    for (auto* w : pv.objs.Roots) delete w;
    62→    62→    pv.objs = M2God::FormObjects();
    63→    63→    pv.byCtrl.clear();
    64→    64→    pv.ctrlOf.clear();
    65→    65→    pv.ok = true;
    66→    66→    // 1) tüm widget'ları yarat
    67→    67→    for (auto* c : g_model.ctrls) {
    68→    68→        if (_wcsicmp(c->type.c_str(), L"TabPage") == 0) {
    69→    69→            // sayfalar WTabBar::Kids'a girer — kendi başına kökte yer almaz
    70→    70→            M2God::WTabPage* pg = new M2God::WTabPage();
    71→    71→            pv.byCtrl[c] = pg;
    72→    72→            pv.ctrlOf[pg] = c;
    73→    73→            continue;
    74→    74→        }
    75→    75→        M2God::Widget* w = M2God::Create(c->type.c_str());
    76→    76→        if (!w) w = new M2God::WLabel();      // bilinmeyen tip → etiket (asla çökmez)
    77→    77→        pv.byCtrl[c] = w;
    78→    78→        pv.ctrlOf[w] = c;
    79→    79→    }
    80→    80→    // 2) özellikleri uygula (SDK sözleşmesi: editör + üretilen kod aynı yol)
    81→    81→    for (auto& kv : pv.byCtrl) {
    82→    82→        Ctrl* c = kv.first;
    83→    83→        M2God::Widget* w = kv.second;
    84→    84→        w->Name = c->name;
    85→    85→        w->Pos  = ImVec2((float)c->x, (float)c->y);
    86→    86→        w->Size = ImVec2((float)c->w, (float)c->h);
    87→    87→        for (auto& p : c->props) w->Set(p.first.c_str(), p.second.c_str());
    88→    88→    }
    89→    89→    // 3) ebeveyn bağla (model sırası korunur → z-sırası)
    90→    90→    for (auto* c : g_model.ctrls) {
    91→    91→        M2God::Widget* w = pv.byCtrl[c];
    92→    92→        M2God::Widget* pw = c->parent ? pv.byCtrl[c->parent] : nullptr;
    93→    93→        pv.objs.Add(w, pw);
    94→    94→    }
    95→    95→}

   100→   100→static ImVec2 ContainerContentOrigin(Ctrl* cont, const ImVec2& contAbs) {
   101→   101→    std::wstring t = Lower(cont->type);
   102→   102→    if (t == L"groupbox")         return ImVec2(contAbs.x + 10, contAbs.y + 10);
   103→   103→    if (t == L"childwindow")      return ImVec2(contAbs.x + 8,  contAbs.y + 8);
   104→   104→    if (t == L"tabpage")          return ImVec2(contAbs.x, contAbs.y);
   105→   105→    if (t == L"collapsingheader") return ImVec2(contAbs.x, contAbs.y + 22);
   106→   106→    return contAbs;
   107→   107→}

   112→   112→static Ctrl* Hit(int fx, int fy) {
   113→   113→    for (int i = (int)g_model.ctrls.size() - 1; i >= 0; i--) {
   114→   114→        Ctrl* c = g_model.ctrls[i];
   115→   115→        if (c->nonVisual) continue;
   116→   116→        // kapanık CollapsingHeader içindekiler görünmez
   117→   117→        if (c->parent && _wcsicmp(c->parent->type.c_str(), L"CollapsingHeader") == 0) {
   118→   118→            // varsayılan açık varsayımı: DefaultOpen != False ise görünür
   119→   119→            if (_wcsicmp(c->parent->Get(L"DefaultOpen", L"True").c_str(), L"False") == 0) continue;
   120→   120→        }
   121→   121→        // pasif sayfa içi (TabBar'ın seçili olmayan sayfası) görünmez
   122→   122→        if (c->parent && _wcsicmp(c->parent->type.c_str(), L"TabPage") == 0) {
   123→   123→            Ctrl* tab = c->parent->parent;
   124→   124→            if (tab) {
   125→   125→                auto pages = TabPagesOf(tab);
   126→   126→                int sel = s_pageSel.count(tab) ? s_pageSel[tab] : 0;
   127→   127→                bool active = false;
   128→   128→                for (int k = 0; k < (int)pages.size(); k++)
   129→   129→                    if (pages[k] == c->parent && k == sel) active = true;
   130→   130→                if (!active) continue;
   131→   131→            }
   132→   132→        }
   133→   133→        POINT ap = g_model.AbsPos(c);
   134→   134→        if (fx >= ap.x && fx <= ap.x + c->w && fy >= ap.y && fy <= ap.y + c->h)
   135→   135→            return c;
   136→   136→    }
   137→   137→    return nullptr;
   138→   138→}

   140→   140→// bırakma hedefi konteyner: en derin, kendisi/yakını DEĞIL, içinde nokta
   141→   141→static Ctrl* DropContainer(int fx, int fy, Ctrl* exclude) {
   142→   142→    Ctrl* best = nullptr;
   143→   143→    double bestArea = 1e18;
   144→   144→    for (auto* c : g_model.ctrls) {
   145→   145→        if (c == exclude || c->nonVisual) continue;
   146→   146→        if (!IsContainerType(c->type)) continue;
   147→   147→        if (exclude && g_model.IsAncestorOf(exclude, c)) continue;
   148→   148→        if (_wcsicmp(c->type.c_str(), L"TabPage") == 0) continue;   // sayfa TabBar üstünden çözülür
   149→   149→        POINT ap = g_model.AbsPos(c);
   150→   150→        if (fx >= ap.x && fx <= ap.x + c->w && fy >= ap.y && fy <= ap.y + c->h) {
   151→   151→            double area = (double)c->w * c->h;
   152→   152→            if (area < bestArea) { bestArea = area; best = c; }
   153→   153→        }
   154→   154→    }
   155→   155→    if (best && _wcsicmp(best->type.c_str(), L"TabBar") == 0) {
   156→   156→        // sekme alanına bırakıldı → AKTİF sayfaya ebeveynlen
   157→   157→        auto pages = TabPagesOf(best);
   158→   158→        int sel = s_pageSel.count(best) ? s_pageSel[best] : 0;
   159→   159→        if (sel < (int)pages.size()) best = pages[sel];
   160→   160→    }
   161→   161→    return best;
   162→   162→}

   174→   174→static void HandleRects(const Rect& r, Rect out[8]) {
   175→   175→    // TL, TR, BL, BR, L, R, T, B
   176→   176→    out[0] = { r.x0 - HANDLE, r.y0 - HANDLE, r.x0 + HANDLE, r.y0 + HANDLE };
   177→   177→    out[1] = { r.x1 - HANDLE, r.y0 - HANDLE, r.x1 + HANDLE, r.y0 + HANDLE };
   178→   178→    out[2] = { r.x0 - HANDLE, r.y1 - HANDLE, r.x0 + HANDLE, r.y1 + HANDLE };
   179→   179→    out[3] = { r.x1 - HANDLE, r.y1 - HANDLE, r.x1 + HANDLE, r.y1 + HANDLE };
   180→   180→    out[4] = { r.x0 - HANDLE, r.y0 + (r.y1 - r.y0) / 2 - HANDLE, r.x0 + HANDLE, r.y0 + (r.y1 - r.y0) / 2 + HANDLE };
   181→   181→    out[5] = { r.x1 - HANDLE, r.y0 + (r.y1 - r.y0) / 2 - HANDLE, r.x1 + HANDLE, r.y0 + (r.y1 - r.y0) / 2 + HANDLE };
   182→   182→    out[6] = { r.x0 + (r.x1 - r.x0) / 2 - HANDLE, r.y0 - HANDLE, r.x0 + (r.x1 - r.x0) / 2 + HANDLE, r.y0 + HANDLE };
   183→   183→    out[7] = { r.x0 + (r.x1 - r.x0) / 2 - HANDLE, r.y1 - HANDLE, r.x0 + (r.x1 - r.x0) / 2 + HANDLE, r.y1 + HANDLE };
   184→   184→}
   185→   185→static bool InRect(const Rect& r, const ImVec2& p) {
   186→   186→    return p.x >= r.x0 && p.x <= r.x1 && p.y >= r.y0 && p.y <= r.y1;
   187→   187→}
   188→   188→static void DrawHandles(ImDrawList* dl, const Rect& r, ImU32 accent) {
   189→   189→    static const ImU32 white = IM_COL32(255, 255, 255, 255);
   190→   190→    static const ImU32 black = IM_COL32(40, 40, 40, 255);
   191→   191→    dl->AddRectFilled(ImVec2(r.x0, r.y0), ImVec2(r.x1, r.y1), accent & IM_COL32_A_MASK, 0.0f);   // dolgu
   192→   192→    dl->AddRect(ImVec2(r.x0, r.y0), ImVec2(r.x1, r.y1), white, 0.0f, 0, 1.0f);
   193→   193→    Rect hr[8];
   194→   194→    HandleRects(r, hr);
   195→   195→    for (auto& h : hr) {
   196→   196→        dl->AddRectFilled(ImVec2(h.x0, h.y0), ImVec2(h.x1, h.y1), white);
   197→   197→        dl->AddRect(ImVec2(h.x0, h.y0), ImVec2(h.x1, h.y1), black, 0.0f, 0, 1.0f);
   198→   198→    }
   199→   199→}

   204→   204→static json::Value CtrlToJson(const Ctrl* c) {
   205→   205→    json::Value o = json::Value::MakeObj();
   206→   206→    o.Set(L"type", c->type);
   207→   207→    o.Set(L"x", (double)c->x); o.Set(L"y", (double)c->y);
   208→   208→    o.Set(L"w", (double)c->w); o.Set(L"h", (double)c->h);
   209→   209→    json::Value p = json::Value::MakeObj();
   210→   210→    for (auto& kv : c->props) p.Set(kv.first, kv.second);
   211→   211→    o.Set(L"props", p);
   212→   212→    json::Value ev = json::Value::MakeObj();
   213→   213→    for (auto& kv : c->events) ev.Set(kv.first, kv.second);
   214→   214→    o.Set(L"events", ev);
   215→   215→    json::Value kids = json::Value::MakeArr();
   216→   216→    for (auto* k : g_model.ChildrenOf(const_cast<Ctrl*>(c)))
   217→   217→        kids.Push(CtrlToJson(k));
   218→   218→    o.Set(L"children", kids);
   219→   219→    return o;
   220→   220→}

   238→   238→static Ctrl* PasteTree(const json::Value& o, Ctrl* parent, int dx, int dy) {
   239→   239→    std::wstring type = o.Get(L"type").AsStr();
   240→   240→    Ctrl* c = g_model.Add(type, 0, 0, parent, false);
   241→   241→    if (!c) return nullptr;
   242→   242→    c->x = (int)o.Get(L"x").AsNum() + dx;
   243→   243→    c->y = (int)o.Get(L"y").AsNum() + dy;
   244→   244→    c->w = (int)o.Get(L"w").AsNum();
   245→   245→    c->h = (int)o.Get(L"h").AsNum();
   246→   246→    const json::Value& p = o.Get(L"props");
   247→   247→    if (p.IsObj()) for (auto& kv : p.obj) c->props[kv.first] = kv.second.AsStr();
   248→   248→    wchar_t b[64]; swprintf(b, 64, L"%d, %d", c->x, c->y); c->props[L"Location"] = b;
   249→   249→    swprintf(b, 64, L"%d, %d", c->w, c->h); c->props[L"Size"] = b;
   250→   250→    const json::Value& ev = o.Get(L"events");
   251→   251→    if (ev.IsObj()) for (auto& kv : ev.obj) c->events[kv.first] = kv.second.AsStr();
   252→   252→    // çocuksa ebeveyn içi geometri; özelse kökte konum
   253→   253→    const json::Value& kids = o.Get(L"children");
   254→   254→    if (kids.IsArr())
   255→   255→        for (auto& k : kids.arr) PasteTree(k, c, 0, 0);
   256→   256→    return c;
   257→   257→}
   258→   258→bool PasteAt(int fx, int fy) {
   259→   259→    if (g_clip.empty()) return false;
   260→   260→    bool ok = false;
   261→   261→    json::Value v = json::Parse(g_clip, &ok);
   262→   262→    if (!ok) return false;
   263→   263→    g_model.PushUndo();
   264→   264→    Ctrl* parent = DropContainer(fx, fy, nullptr);
   265→   265→    ImVec2 origin(0, 0);
   266→   266→    int px = fx, py = fy;
   267→   267→    if (parent) {
   268→   268→        POINT ap = g_model.AbsPos(parent);
   269→   269→        ImVec2 co = ContainerContentOrigin(parent, ImVec2((float)ap.x, (float)ap.y));
   270→   270→        px = (int)(fx - co.x);
   271→   271→        py = (int)(fy - co.y);
   272→   272→    }
   273→   273→    Ctrl* c = PasteTree(v, parent, px, py);
   274→   274→    if (c) {
   275→   275→        g_model.selected = c;
   276→   276→        g_model.formSelected = false;
   277→   277→        Changed();
   278→   278→        Status(L"Yapıştırıldı: " + c->name);
   279→   279→        return true;
   280→   280→    }
   281→   281→    return false;
   282→   282→}

   296→   296→static void DrawTabStrip(ImDrawList* dl, Ctrl* tab, const ImVec2& abs, Preview& pv,
   297→   297→                         ImVec2& pageOrigin, ImVec2& pageSize) {
   298→   298→    auto pages = TabPagesOf(tab);
   299→   299→    int sel = s_pageSel.count(tab) ? s_pageSel[tab] : 0;
   300→   300→    if (sel >= (int)pages.size()) sel = (int)pages.size() - 1;
   301→   301→    if (sel < 0) sel = 0;
   302→   302→    s_pageSel[tab] = sel;
   303→   303→    ImU32 bgDim   = ImGui::GetColorU32(ImGuiCol_TitleBg);
   304→   304→    ImU32 bgActive = ImGui::GetColorU32(ImGuiCol_TitleBgActive);
   305→   305→    ImU32 bgIdle  = ImGui::GetColorU32(ImGuiCol_TabUnfocused, 0.85f);
   306→   306→    ImU32 bgHover = ImGui::GetColorU32(ImGuiCol_TabUnfocusedActive, 0.9f);
   307→   307→    ImU32 textCol = ImGui::GetColorU32(ImGuiCol_Text);
   308→   308→    ImVec2 stripMax(abs.x + tab->w, abs.y + TAB_STRIP_H);
   309→   309→    dl->AddRectFilled(abs, stripMax, bgDim);
   310→   310→    dl->AddRect(abs, ImVec2(abs.x + tab->w, abs.y + tab->h), ImGui::GetColorU32(ImGuiCol_Border), 4.0f);
   311→   311→    float tx = abs.x + 4;
   312→   312→    for (int i = 0; i < (int)pages.size(); i++) {
   313→   313→        std::wstring t = pages[i]->Get(L"Text");
   314→   314→        if (t.empty()) t = pages[i]->name;
   315→   315→        ImVec2 ts = ImGui::CalcTextSize(M2God::W2U(t).c_str());
   316→   316→        float w = ts.x + 18.0f;
   317→   317→        ImVec2 a(tx, abs.y + 3), b(tx + w, abs.y + TAB_STRIP_H);
   318→   318→        dl->AddRectFilled(a, b, i == sel ? bgActive : bgIdle, 4.0f, ImDrawFlags_RoundCornersTop);
   319→   319→        dl->AddText(ImVec2(a.x + 9, a.y + (TAB_STRIP_H - 3 - ts.y) * 0.5f), textCol, M2God::W2U(t).c_str());
   320→   320→        // sekme tıklaması (tasarım etkileşimi — form kilidiyle)
   321→   321→        ImVec2 mp = ImGui::GetIO().MousePos;
   322→   322→        if (mp.x >= a.x && mp.x <= b.x && mp.y >= a.y && mp.y <= b.y) {
   323→   323→            if (i != sel) dl->AddRect(a, b, bgHover, 4.0f, ImDrawFlags_RoundCornersTop, 1.5f);
   324→   324→            if (ImGui::IsMouseClicked(ImGuiMouseButton_Left))
   325→   325→                s_pageSel[tab] = i;
   326→   326→        }
   327→   327→        tx += w;
   328→   328→    }
   329→   329→    pageOrigin = ImVec2(abs.x + 4, abs.y + TAB_STRIP_H + 4);
   330→   330→    pageSize   = ImVec2((float)tab->w - 8, (float)tab->h - TAB_STRIP_H - 8);
   331→   331→}

   336→   336→static void DrawComponentTray(ImDrawList* dl, const ImVec2& origin, const ImVec2& size) {
   337→   337→    dl->AddRectFilled(origin, ImVec2(origin.x + size.x, origin.y + size.y),
   338→   338→                      ImGui::GetColorU32(ImGuiCol_TabUnfocused), 4.0f);
   339→   339→    float x = origin.x + 8;
   340→   340→    float cy = origin.y + size.y * 0.5f;
   341→   341→    dl->AddText(ImVec2(x, cy - 26), ImGui::GetColorU32(ImGuiCol_TextDisabled), "Bilesenler");
   342→   342→    x += 90;
   343→   343→    for (auto* c : g_model.ctrls) {
   344→   344→        if (!c->nonVisual) continue;
   345→   345→        std::string nm = M2God::W2U(c->name + L" (" + c->type + L")");
   346→   346→        ImVec2 ts = ImGui::CalcTextSize(nm.c_str());
   347→   347→        ImVec2 a(x, cy - ts.y * 0.5f - 3), b(x + ts.x + 16, cy + ts.y * 0.5f + 3);

   369→   369→static void RenderPreviewTree(Preview& pv, Ctrl* tabRoot, const ImVec2& origin) {
   370→   370→    // 1) görsel olmayanlar yok; TabBar özel çizilir
   371→   371→    for (auto* w : pv.objs.Roots) {

   377→   377→            ImDrawList* dl = ImGui::GetWindowDrawList();

   380→   380→            auto pages = TabPagesOf(c);
   381→   381→            int sel = s_pageSel.count(c) ? s_pageSel[c] : 0;
   382→   382→            if (sel >= (int)pages.size()) sel = (int)pages.size() - 1;
   383→   383→            // içerik: seçili sayfanın çocukları (SDK yol — WTabPage::RenderKids)
   384→   384→            if (sel >= 0 && sel < (int)pages.size()) {
   385→   385→                auto itW = pv.byCtrl.find(pages[sel]);
   386→   386→                if (itW != pv.byCtrl.end()) {
   387→   387→                    M2God::WTabPage* pg = (M2God::WTabPage*)itW->second;
   388→   388→                    ImVec2 pgOrigin(abs.x + 4, abs.y + TAB_STRIP_H + 4);
   389→   389→                    // çerçeve
   390→   390→                    dl->AddRectFilled(ImVec2(abs.x + 2, abs.y + TAB_STRIP_H + 2),
   391→   391→                                      ImVec2(abs.x + c->w - 2, abs.y + c->h - 2),
   392→   392→                                      ImGui::GetColorU32(ImGuiCol_WindowBg, 0.4f));
   393→   393→                    pg->RenderKids(pgOrigin);
   394→   394→                }
   395→   395→            }
   396→   396→            // şerit en üstte (sekmeler tıklanabilir)
   397→   397→            ImVec2 po, ps;
   398→   398→            DrawTabStrip(dl, c, abs, pv, po, ps);
   399→   399→            continue;
   400→   400→        }
   401→   401→        w->Render(origin);
   402→   402→    }
   403→   403→}

   668→   668→static void BeginDrag(DragMode m, Ctrl* c) {
   669→   669→    s_drag.mode = m;
   670→   670→    s_drag.ctrl = c;
   671→   671→    s_drag.moved = false;
   672→   672→    s_drag.undoPushed = false;
   673→   673→    if (c) {
   674→   674→        s_drag.sx = c->x; s_drag.sy = c->y;
   675→   675→        s_drag.sw = c->w; s_drag.sh = c->h;
   676→   676→    } else {
   677→   677→        s_drag.fsw = g_model.formW;
   678→   678→        s_drag.fsh = g_model.formH;
   679→   679→    }

   681→   681→static void Snap(int& v) { if (g_snap) v = ((v + GRID_SIZE / 2) / GRID_SIZE) * GRID_SIZE; }

   683→   683→static void UpdateDrag(const ImVec2& mp, const ImVec2& paper) {
   684→   684→    Ctrl* c = s_drag.ctrl;
   685→   685→    if (!s_drag.undoPushed) {
   686→   686→        if (s_drag.mode == DR_MOVE || s_drag.mode >= DR_SZ_TL) {
   687→   687→            g_model.PushUndo();      // ilk gerçek hareket öncesi kayıt
   688→   688→            s_drag.undoPushed = true;
   689→   689→        }
   690→   690→    }
   691→   691→    s_drag.moved = true;
   692→   692→    int fx = (int)(mp.x - paper.x), fy = (int)(mp.y - paper.y);
   693→   693→    if (!c) {     // form boyutlandırma
   694→   694→        int W = fx, H = fy;
   695→   695→        switch (s_drag.mode) {
   696→   696→        case DR_SZ_TL: case DR_SZ_TR: case DR_SZ_BL: case DR_SZ_BR: W = fx; H = fy; break;
   697→   697→        case DR_SZ_L: W = fx; H = g_model.formH; break;
   698→   698→        case DR_SZ_R: W = fx; H = g_model.formH; break;
   699→   699→        case DR_SZ_T: W = g_model.formW; H = fy; break;
   700→   700→        case DR_SZ_B: W = g_model.formW; H = fy; break;
   701→   701→        default: return;
   702→   702→        }
   703→   703→        if (W < 60) W = 60;
   704→   704→        if (H < 40) H = 40;
   705→   705→        Snap(W); Snap(H);
   706→   706→        g_model.formW = W;
   707→   707→        g_model.formH = H;
   708→   708→        wchar_t b[64]; swprintf(b, 64, L"%d, %d", W, H);
   709→   709→        g_model.formProps[L"ClientSize"] = b;
   710→   710→        g_modelDirty = true;
   711→   711→        return;
   712→   712→    }
   713→   713→    // taşıma: mutlak konum → ebeveyn görelisi
   714→   714→    if (s_drag.mode == DR_MOVE) {
   715→   715→        int ax = (int)(mp.x - paper.x - s_drag.grabOff.x);
   716→   716→        int ay = (int)(mp.y - paper.y - s_drag.grabOff.y);
   717→   717→        if (c->parent) {
   718→   718→            POINT ap = g_model.AbsPos(c->parent);
   719→   719→            ImVec2 co = ContainerContentOrigin(c->parent, ImVec2((float)ap.x, (float)ap.y));
   720→   720→            ax = (int)(ax - co.x);
   721→   721→            ay = (int)(ay - co.y);
   722→   722→            if (ax < -c->w + 6) ax = -c->w + 6;      // ebeveyn içinde en az 6px görünsün
   723→   723→            if (ay < -c->h + 6) ay = -c->h + 6;
   724→   724→        }
   725→   725→        Snap(ax); Snap(ay);
   726→   726→        if (ax != c->x || ay != c->y) {
   727→   727→            c->x = ax; c->y = ay;
   728→   728→            wchar_t b[64]; swprintf(b, 64, L"%d, %d", ax, ay);
   729→   729→            c->props[L"Location"] = b;
   730→   730→            g_modelDirty = true;
   731→   731→        }
   732→   732→        return;
   733→   733→    }
   734→   734→    if (s_drag.mode >= DR_SZ_TL) {
   735→   735→        // genel boyutlandırma: başlangıç kutusu + delta (mutlak form koordinatında)
   736→   736→        int x0 = s_drag.sx, y0 = s_drag.sy;         // ebeveyn görelisi başlangıç
   737→   737→        POINT ap0 = g_model.AbsPos(c);
   738→   738→        int X0 = ap0.x, Y0 = ap0.y;                // başlangıç mutlak (form) kutusu
   739→   739→        int X1 = X0 + s_drag.sw, Y1 = Y0 + s_drag.sh;
   740→   740→        switch (s_drag.mode) {
   741→   741→        case DR_SZ_TL: X0 = fx; Y0 = fy; break;
   742→   742→        case DR_SZ_TR: X1 = fx; Y0 = fy; break;
   743→   743→        case DR_SZ_BL: X0 = fx; Y1 = fy; break;
   744→   744→        case DR_SZ_BR: X1 = fx; Y1 = fy; break;
   745→   745→        case DR_SZ_L:  X0 = fx; break;
   746→   746→        case DR_SZ_R:  X1 = fx; break;
   747→   747→        case DR_SZ_T:  Y0 = fy; break;
   748→   748→        case DR_SZ_B:  Y1 = fy; break;
   749→   749→        default: break;
   750→   750→        }
   751→   751→        (void)x0; (void)y0;
   752→   752→        if (X1 - X0 < 4) X1 = X0 + 4;
   753→   753→        if (Y1 - Y0 < 4) Y1 = Y0 + 4;
   754→   754→        // ebeveyne geri çevir
   755→   755→        int nx = X0, ny = Y0, nw = X1 - X0, nh = Y1 - Y0;
   756→   756→        if (c->parent) {
   757→   757→            POINT pap = g_model.AbsPos(c->parent);
   758→   758→            ImVec2 co = ContainerContentOrigin(c->parent, ImVec2((float)pap.x, (float)pap.y));
   759→   759→            nx = (int)(X0 - co.x);
   760→   760→            ny = (int)(Y0 - co.y);
   761→   761→        }
   762→   762→        c->x = nx; c->y = ny; c->w = nw; c->h = nh;
   763→   763→        wchar_t b[64];
   764→   764→        swprintf(b, 64, L"%d, %d", nx, ny); c->props[L"Location"] = b;
   765→   765→        swprintf(b, 64, L"%d, %d", nw, nh); c->props[L"Size"] = b;
   766→   766→        g_modelDirty = true;
   767→   767→    }
   768→   768→}
   769→   769→static void EndDrag() {
   770→   770→    if (s_drag.moved) {
   771→   771→        Changed();
   772→   772→        if (s_drag.ctrl) Status(L"Taşındı: " + s_drag.ctrl->name);
   773→   773→        else Status(L"ClientSize: " + std::to_wstring(g_model.formW) + L", " + std::to_wstring(g_model.formH));
   774→   774→    }
   775→   775→    s_drag.mode = DR_NONE;
   776→   776→    s_drag.ctrl = nullptr;
   777→   777→}
   778→   778→static void CancelDrag() {
   779→   779→    if (s_drag.ctrl) {
   780→   780→        s_drag.ctrl->x = s_drag.sx; s_drag.ctrl->y = s_drag.sy;
   781→   781→        s_drag.ctrl->w = s_drag.sw; s_drag.ctrl->h = s_drag.sh;
   782→   782→        wchar_t b[64];
   783→   783→        swprintf(b, 64, L"%d, %d", s_drag.ctrl->x, s_drag.ctrl->y);
   784→   784→        s_drag.ctrl->props[L"Location"] = b;
   785→   785→        swprintf(b, 64, L"%d, %d", s_drag.ctrl->w, s_drag.ctrl->h);
   786→   786→        s_drag.ctrl->props[L"Size"] = b;
   787→   787→    } else {
   788→   788→        g_model.formW = s_drag.fsw; g_model.formH = s_drag.fsh;
   789→   789→        wchar_t b[64]; swprintf(b, 64, L"%d, %d", g_model.formW, g_model.formH);
   790→   790→        g_model.formProps[L"ClientSize"] = b;
   791→   791→    }
   792→   792→    s_drag.mode = DR_NONE;
   793→   793→    s_drag.ctrl = nullptr;
   794→   794→    Status(L"İptal edildi");
   795→   795→}
