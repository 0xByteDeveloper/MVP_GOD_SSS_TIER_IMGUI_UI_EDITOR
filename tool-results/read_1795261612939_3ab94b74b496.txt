    17→static std::wstring EscCppStr(const std::wstring& s) {
    18→    std::wstring e;
    19→    for (wchar_t ch : s) {
    20→        if (ch == L'\\') e += L"\\\\";
    21→        else if (ch == L'"') e += L"\\\"";
    22→        else if (ch == L'\n') e += L"\\n";
    23→        else if (ch == L'\r') e += L"\\r";
    24→        else if (ch == L'\t') e += L"\\t";
    25→        else e += ch;
    26→    }
    27→    return e;
    28→}

    30→std::wstring IdMacroOf(const std::wstring& ctrlName) {
    31→    std::wstring up = ctrlName;
    32→    for (auto& ch : up) {
    33→        if (iswalnum(ch)) ch = (wchar_t)towupper(ch);
    34→        else ch = L'_';
    35→    }
    36→    return L"IDC_" + up;
    37→}

    56→// şablon metnini belirtilen form adına çevir
    57→static std::wstring TmplFix(std::wstring t, const std::wstring& name) {
    58→    if (name.empty() || name == L"Form1") return t;   // Form1 = birebir eski davranış
    59→    std::wstring ns = L"M2God" + name;                 // M2GodForm2...
    60→    t = ReplaceAll(t, L"M2GodForm1", ns);               // pencere sınıf dizisi (L"M2GodForm1")
    61→    t = ReplaceAll(t, L"M2GodForm::", ns + L"::");
    62→    t = ReplaceAll(t, L"namespace M2GodForm", L"namespace " + ns);
    63→    t = ReplaceAll(t, L"Form1_WndProc", name + L"_WndProc");
    64→    t = ReplaceAll(t, L"Form1.cpp", name + L".cpp");
    65→    t = ReplaceAll(t, L"Form1.h", name + L".h");
    66→    return t;
    67→}

    88→static bool StripWinMain(std::wstring& t, const std::wstring& repl) {
    89→    size_t b = t.find(L"int APIENTRY wWinMain");
    90→    size_t e = t.find(L"//== M2GOD:WNDPROC-END");
    91→    if (b == std::wstring::npos || e == std::wstring::npos || e < b) return false;
    92→    t = t.substr(0, b) + repl + t.substr(e);
    93→    return true;
    94→}

   162→        std::wstring ti = c->Get(L"Title");   if (!ti.empty()) o << L"    " << c->name << L"->Title = L\"" << EscCppStr(ti) << L"\";\n";
   163→        std::wstring id = c->Get(L"InitialDirectory"); if (!id.empty()) o << L"    " << c->name << L"->InitialDirectory = L\"" << EscCppStr(id) << L"\";\n";
   164→        if (BoolLit(c->Get(L"Multiselect")) != L"false")
   165→            o << L"    " << c->name << L"->Multiselect = " << BoolLit(c->Get(L"Multiselect")) << L";\n";

   167→            std::wstring de = c->Get(L"DefaultExt"); if (!de.empty()) o << L"    " << c->name << L"->DefaultExt = L\"" << EscCppStr(de) << L"\";\n";

   169→                o << L"    " << c->name << L"->OverwritePrompt = false;\n";

   171→        return o.str();
   172→    }
   173→    if (t == L"ColorDialog") {
   174→        o << L"    " << c->name << L" = new M2God::ColorDialog();\n";
   175→        if (_wcsicmp(c->Get(L"FullOpen", L"False").c_str(), L"true") == 0)
   176→            o << L"    " << c->name << L"->FullOpen = true;\n";

   250→static std::wstring EmitCreateCall(const Ctrl* c) {
   251→    std::wostringstream o;
   252→    if (c->type == L"Timer") {
   253→        o << L"    " << c->name << L" = new M2God::Timer(hWnd, " << IdMacroOf(c->name) << L");\n";
   254→        o << L"    " << c->name << L"->Interval = " << c->Get(L"Interval", L"100") << L";\n";
   255→        if (_wcsicmp(c->Get(L"Enabled", L"False").c_str(), L"true") == 0)
   256→            o << L"    " << c->name << L"->Enabled = true;   // tasarımda Enabled=True\n";
   257→        else
   258→            o << L"    // " << c->name << L"->Enabled = true;   // Timer'ı başlatmak için\n";
   259→        return o.str();
   260→    }
   261→    if (!M2ComponentTypeOf(c->type).empty()) return EmitComponentCreate(c);
   262→    if (c->nonVisual) return EmitComponentCreate(c);
   263→    // --- FAZ 2.6: TabPage — tabControl1->TabPages->Add(L"...", ID_) ---
   264→    if (c->type == L"TabPage") {
   265→        std::wstring tabName = c->parent ? c->parent->name : std::wstring(L"tabControl1");
   266→        o << L"    // sayfa: " << c->name << L" (" << tabName << L")\n";
   267→        o << L"    " << c->name << L" = " << tabName << L"->TabPages->Add(L\""
   268→          << EscCppStr(c->Get(L"Text")) << L"\", " << IdMacroOf(c->name) << L");\n";
   269→        {
   270→            std::wstring bv = c->Get(L"BackColor");
   271→            if (!bv.empty() && _wcsicmp(bv.c_str(), L"Control") != 0) {
   272→                COLORREF cr;
   273→                if (ParseColor(bv, cr))
   274→                    o << L"    " << c->name << L"->BackColor = RGB(" << (int)GetRValue(cr) << L", "
   275→                      << (int)GetGValue(cr) << L", " << (int)GetBValue(cr) << L");\n";
   276→            }
   277→            std::wstring fv = c->Get(L"ForeColor");
   278→            if (!fv.empty() && _wcsicmp(fv.c_str(), L"ControlText") != 0) {
   279→                COLORREF cr;
   280→                if (ParseColor(fv, cr))
   281→                    o << L"    " << c->name << L"->ForeColor = RGB(" << (int)GetRValue(cr) << L", "
   282→                      << (int)GetGValue(cr) << L", " << (int)GetBValue(cr) << L");\n";
   283→            }
   284→        }
   285→        if (_wcsicmp(c->Get(L"Enabled", L"True").c_str(), L"false") == 0)
   286→            o << L"    " << c->name << L"->Enabled = false;\n";
   287→        { std::wstring tg = c->Get(L"Tag"); if (!tg.empty())
   288→            o << L"    " << c->name << L"->Tag = L\"" << EscCppStr(tg) << L"\";\n"; }
   289→        { std::wstring tt = c->Get(L"ToolTipText"); if (!tt.empty())
   290→            o << L"    " << c->name << L"->ToolTipText = L\"" << EscCppStr(tt) << L"\";\n"; }
   291→        return o.str();
   292→    }
   293→    factory::Win32Class m = factory::MapClass(c->type);
   294→    if (m.cls.empty()) return EmitComponentCreate(c);
   295→    std::wstring parentVar = c->parent ? (c->parent->name + L"->Handle") : std::wstring(L"hWnd");
   296→    if (c->parent)
   297→        o << L"    // " << c->name << L" → " << c->parent->name << L" içine (child)\n";
   298→    std::wstring txt = c->Get(L"Text");
   299→    std::wstring esc = EscCppStr(txt);
   300→    bool isTabCtrl = (c->type == L"TabControl");
   301→    std::wstring ctorText = isTabCtrl ? L"" : esc;   // TabControl sayfaları Pages->Add ile eklenir
   302→    std::wstring ctorArgs = parentVar + L", " + IdMacroOf(c->name) + L", L\"" + ctorText + L"\", " +
   303→                            std::to_wstring(c->x) + L", " + std::to_wstring(c->y) + L", " +
   304→                            std::to_wstring(c->w) + L", " + std::to_wstring(c->h);
   305→    // TextBox ailesi: multiline parametresi
   306→    if (c->type == L"TextBox" || c->type == L"MaskedTextBox" || c->type == L"NumericUpDown") {
   307→        bool ml = (c->type == L"TextBox") && _wcsicmp(c->Get(L"MultiLine", L"False").c_str(), L"true") == 0;
   308→        ctorArgs += ml ? L", true" : L", false";
   309→    }
   310→    o << L"    " << c->name << L" = new M2God::" << M2TypeOf(c->type) << L"(" << ctorArgs << L");\n";
   311→    // --- özellik emisyonu ---
   312→    // Items (ListBox/CheckedListBox/ComboBox): her satır bir öğe
   313→    // (ListView kendi bloğunda '|' ile hücre ayrımıyla üretilir)
   314→    auto itItems = c->props.find(L"Items");
   315→    if (c->type != L"ListView" && itItems != c->props.end() && !itItems->second.empty()) {
   316→        o << L"    " << c->name << L"->Items->Clear();\n";
   317→        for (auto& line : Split(itItems->second, L'\n')) {
   318→            std::wstring t = Trim(line);
   319→            if (!t.empty())
   320→                o << L"    " << c->name << L"->Items->Add(L\"" << EscCppStr(t) << L"\");\n";
   321→        }
   322→    }
   323→    // FAZ 2.6.3: ListView Sütunlar (tasarımcı "Columns": her satır "Başlık, Genişlik")
   324→    if (c->type == L"ListView") {
   325→        auto itCols = c->props.find(L"Columns");
   326→        if (itCols != c->props.end() && !itCols->second.empty()) {
   327→            o << L"    " << c->name << L"->Columns->Clear();\n";

   346→                std::wstring t = Trim(line);
   347→                if (t.empty()) continue;
   348→                auto cells = Split(t, L'|');

   384→                        while (pos < t.size() && t[pos] == L' ') { sp++; pos++; }
   385→                        level += sp / 4;
   386→                    } else break;

   397→                for (int k = level + 1; k < 16; k++) lastAtLevel[k] = -1;
   398→            }
   399→        }

   786→    std::wostringstream h;

   790→            continue;
   791→        }
   792→        std::wstring wrapper = M2ComponentTypeOf(c->type);
   793→        if (!wrapper.empty()) {
   794→            // SDK sarmalayıcısı olan bileşen/diyalog (Faz 2.5)
   795→        } else if (c->nonVisual) {

   798→            factory::Win32Class mc = factory::MapClass(c->type);
   799→            if (mc.cls.empty()) continue;

   802→        h << (asExtern ? L"extern " : L"") << L"M2God::" << wrapper
   803→          << L"* " << c->name << (asExtern ? L";\n" : L" = nullptr;\n");
   804→    }
   805→    return h.str();
   806→}

   808→// WM_COMMAND bildirim eşlemesi (Faz 2.5 — boş dönerse WM_NOTIFY/WM_HSCROLL'a bakılır)
   809→static std::wstring CmdNotifyOf(const std::wstring& type, const std::wstring& evName) {
   810→    if (evName == L"Click" || evName == L"CheckedChanged") {
   811→        if (type == L"Label" || type == L"LinkLabel" || type == L"PictureBox") return L"STN_CLICKED";
   812→        return L"BN_CLICKED";
   813→    }
   814→    if (evName == L"TextChanged") {
   815→        if (type == L"ComboBox") return L"CBN_EDITCHANGE";
   816→        return L"EN_CHANGE";
   817→    }
   818→    if (evName == L"SelectedIndexChanged") {
   819→        if (type == L"ComboBox") return L"CBN_SELCHANGE";
   820→        if (type == L"ListBox") return L"LBN_SELCHANGE";
   821→        return L"";   // ListView/CheckedListBox/TabControl → WM_NOTIFY
   822→    }
   823→    if (evName == L"DoubleClick") {
   824→        if (type == L"ListBox") return L"LBN_DBLCLK";
   825→        if (type == L"ComboBox") return L"CBN_DBLCLK";
   826→        if (type == L"Label" || type == L"LinkLabel" || type == L"PictureBox") return L"STN_DBLCLK";
   827→        return L"BN_DOUBLECLICKED";
   828→    }
   829→    if (evName == L"ValueChanged") {
   830→        if (type == L"NumericUpDown") return L"EN_CHANGE";
   831→        return L"";
   832→    }
   833→    return L"";
   834→}

  1389→static bool ReplaceRegion(std::wstring& text, const wchar_t* beginMark, const wchar_t* endMark, const std::wstring& newContent) {
  1390→    size_t b = text.find(beginMark);
  1391→    if (b == std::wstring::npos) return false;
  1392→    size_t e = text.find(endMark, b);
  1393→    if (e == std::wstring::npos) return false;
  1394→    size_t contentStart = b + wcslen(beginMark);
  1395→    text = text.substr(0, contentStart) + L"\n" + newContent + text.substr(e);
  1396→    return true;
  1397→}

  1422→int EnsureHandler(const std::wstring& cppPath, const std::wstring& hPath,
  1423→                  const std::wstring& handlerName, const std::wstring& declLine) {
  1424→    std::string tmp8;
  1425→    if (!ReadFileText(cppPath, tmp8)) return 0;
  1426→    std::wstring cpp = Utf8ToWide(tmp8);
  1427→    std::wstring sig = L"void " + handlerName + L"(";
  1428→    size_t found = Lower(cpp).find(Lower(sig));
  1429→    if (found == std::wstring::npos) {
  1430→        // stub ekle
  1431→        std::wstring stub = L"void " + handlerName + L"(HWND hWnd) {\n    // TODO: " + handlerName + L"\n}\n\n";
  1432→        const wchar_t* endMark = L"//== M2GOD:EVENTS-END";
  1433→        size_t e = cpp.find(endMark);
  1434→        if (e != std::wstring::npos) {
  1435→            cpp.insert(e, stub);
  1436→        } else {
  1437→            cpp += L"\n" + stub;
  1438→        }
  1439→        WriteFileText(cppPath, WideToUtf8(cpp));
  1440→        found = Lower(cpp).find(Lower(sig));
  1441→    }
  1442→    // bildirimi .h'ye ekle (yoksa)
  1443→    if (!declLine.empty() && FileExists(hPath)) {
  1444→        std::string t8;
  1445→        ReadFileText(hPath, t8);
  1446→        std::wstring h = Utf8ToWide(t8);
  1447→        std::wstring decl = L"    " + declLine + L"\n";
  1448→        size_t dr = Lower(h).find(Lower(L"void " + declLine));
  1449→        if (dr == std::wstring::npos) {
  1450→            const wchar_t* endMark = L"//== M2GOD:EVENTDECL-END";
  1451→            size_t e = h.find(endMark);
  1452→            if (e != std::wstring::npos) h.insert(e, decl);
  1453→            else h += L"\n" + decl;
  1454→            WriteFileText(hPath, WideToUtf8(h));
  1455→        }
  1456→    }
  1457→    if (found == std::wstring::npos) return 0;
  1458→    return LineOfPos(cpp, found);
  1459→}
