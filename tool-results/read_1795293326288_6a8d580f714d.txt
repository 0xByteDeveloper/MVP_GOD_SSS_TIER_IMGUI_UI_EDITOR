    34→std::string OpenDlg(const char* title, const char* filter) {
    35→    char file[MAX_PATH] = "";
    36→    OPENFILENAMEA ofn = { sizeof(ofn) };
    37→    ofn.hwndOwner = GetActiveWindow();
    38→    ofn.lpstrFile = file;
    39→    ofn.nMaxFile = MAX_PATH;
    40→    ofn.lpstrTitle = title;
    41→    ofn.lpstrFilter = filter;
    42→    ofn.Flags = OFN_FILEMUSTEXIST | OFN_EXPLORER | OFN_NOCHANGEDIR;
    43→    if (GetOpenFileNameA(&ofn)) return file;
    44→    return "";
    45→}
    46→std::string SaveDlg(const char* title, const char* filter, const char* defExt) {
    47→    char file[MAX_PATH] = "";
    48→    OPENFILENAMEA ofn = { sizeof(ofn) };
    49→    ofn.hwndOwner = GetActiveWindow();
    50→    ofn.lpstrFile = file;
    51→    ofn.nMaxFile = MAX_PATH;
    52→    ofn.lpstrTitle = title;
    53→    ofn.lpstrFilter = filter;
    54→    ofn.lpstrDefExt = defExt;
    55→    ofn.Flags = OFN_OVERWRITEPROMPT | OFN_EXPLORER | OFN_NOCHANGEDIR;
    56→    if (GetSaveFileNameA(&ofn)) return file;
    57→    return "";
    58→}

   174→static void DoSave(bool saveAs) {
   175→    if (saveAs) {
   176→        std::string p = SaveDlg("Proje kaydet", "IMGUI GOD projesi\0*.imgod\0", "imgod");
   177→        if (p.empty()) return;
   178→        if (forms::SaveProjectAs(p)) s_status = "Kaydedildi: " + p;
   179→        else s_status = "Kaydedilemedi";
   180→        return;
   181→    }
   182→    if (!forms::ProjectDir().empty()) {
   183→        if (forms::SaveFolderProject(forms::ProjectDir())) s_status = "Kaydedildi: " + forms::ProjectDir();
   184→        else s_status = "Kaydedilemedi";
   185→    } else DoSave(true);
   186→}
   187→static void DoSaveFolder() {
   188→    std::string info = SaveDlg("Klas\303\266r projesi kaydet (Forms.info)", "Forms.info\0Forms.info\0", "info");
   189→    if (info.empty()) return;
   190→    if (forms::SaveFolderProject(DirOfA(info))) s_status = "Klas\303\266r projesi: " + DirOfA(info);
   191→    else s_status = "Kaydedilemedi";
   192→}

   240→static void DrawToolbox(ImVec2 min, ImVec2 max) {
   241→    ImGui::SetNextWindowPos(min);
   242→    ImGui::SetNextWindowSize(ImVec2(max.x - min.x, max.y - min.y));
   243→    ImGui::Begin("Ara\303\247 Kutusu", nullptr, ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoSavedSettings);
   244→    static char search[128] = "";
   245→    ImGui::TextUnformatted("Ara (kategori/nesne):");
   246→    ImGui::SetNextItemWidth(-1);
   247→    ImGui::InputText("##tbs", search, sizeof search);
   248→    ImGui::Separator();
   249→    for (auto& cat : g_catalog.categories) {
   250→        // arama filtresi
   251→        bool show = search[0] == 0;
   252→        for (auto& e : g_catalog.catElements[cat]) {
   253→            std::wstring low = LowerW(Utf8ToWideStr(e));
   254→            std::wstring sw = LowerW(Utf8ToWideStr(std::string(search)));
   255→            if (low.find(sw) != std::wstring::npos || cat.find(search) != std::string::npos) show = true;
   256→        }
   257→        if (!show) continue;
   258→        ImGui::SetNextItemOpen(true, ImGuiCond_Always);
   259→        if (ImGui::TreeNodeEx(cat.c_str(), ImGuiTreeNodeFlags_DefaultOpen)) {
   260→            for (auto& e : g_catalog.catElements[cat]) {
   261→                ImGui::PushID(e.c_str());
   262→                // tık = YERLEŞTİRME MODU (arm — ESC iptal); çift tık = aktif forma direkt ekle
   263→                if (ImGui::Selectable(e.c_str(), canvas::IsArmed() && canvas::g_armTool == e,
   264→                                      ImGuiSelectableFlags_AllowDoubleClick)) {
   265→                    if (ImGui::IsMouseDoubleClicked(0)) {
   266→                        canvas::ArmTool(e);
   267→                        canvas::AddArmedToForm();
   268→                        canvas::CancelArm();
   269→                        s_status = "Eklendi: " + e + " (aktif forma)";
   270→                    } else {
   271→                        canvas::ArmTool(e);
   272→                        s_status = "Yerle\305\237tir: " + e + " — forma t\304\261kla (ESC iptal)";
   273→                    }
   274→                }
   275→                if (ImGui::IsItemHovered()) {
   276→                    const ElementDef* d = g_catalog.Find(e);
   277→                    if (d) ImGui::SetTooltip("%s\n%s\nT\304\261kla: yerle\305\237tir · \303\247ift t\304\261kla: forma ekle",
   278→                                             d->cppApi.c_str(), d->kind.c_str());
   279→                }
   280→                ImGui::PopID();
   281→            }
   282→            ImGui::TreePop();
   283→        }
   284→    }
   285→    ImGui::End();
   286→}

   288→// kod gezgini satırı — SAĞLAM seçim (Selectable) + bağlam menüsü + hiyerarşi
   289→static void DrawCtrlRow(Ctrl* c, int formIdx, int depth) {
   290→    ImGui::PushID(c->name.c_str());
   291→    ImGui::PushID(depth);
   292→    std::string line = c->name + "  :  " + c->type;
   293→    bool sel = g_model.IsSelected(c);
   294→    bool container = IsContainerType(c->type) && !c->nonVisual && c->type != "Menu";
   295→    if (container) {
   296→        ImGuiTreeNodeFlags fl = ImGuiTreeNodeFlags_DefaultOpen | (sel ? ImGuiTreeNodeFlags_Selected : 0);
   297→        bool open = ImGui::TreeNodeEx(line.c_str(), fl);
   298→        if (ImGui::IsItemClicked(0)) {
   299→            if (formIdx != forms::ActiveIndex()) forms::SwitchTo(formIdx);
   300→            canvas::Select(c, ImGui::GetIO().KeyShift || ImGui::GetIO().KeyCtrl);
   301→        }
   302→        if (ImGui::IsItemHovered()) ImGui::SetTooltip("T\304\261kla: se\303\247 · a\303\247/kapa: \303\247ocuklar");
   303→        if (open) {
   304→            for (Ctrl* ch : g_model.ChildrenOf(c)) DrawCtrlRow(ch, formIdx, depth + 1);
   305→            ImGui::TreePop();
   306→        }
   307→    } else {
   308→        if (ImGui::Selectable(line.c_str(), sel)) {
   309→            if (formIdx != forms::ActiveIndex()) forms::SwitchTo(formIdx);
   310→            canvas::Select(c, ImGui::GetIO().KeyShift || ImGui::GetIO().KeyCtrl);
   311→        }
   312→        if (ImGui::IsItemHovered()) ImGui::SetTooltip("T\304\261kla: se\303\247");
   313→    }
   314→    // bağlam menüsü (sağ tık)
   315→    if (ImGui::BeginPopupContextItem("##ctx")) {
   316→        if (ImGui::MenuItem("Se\303\247")) canvas::Select(c, false);
   317→        ImGui::Separator();
   318→        if (ImGui::MenuItem("Kopyala (Ctrl+C)", nullptr, false, g_model.IsSelected(c))) g_model.CopySelection();
   319→        if (ImGui::MenuItem("Kes (Ctrl+X)", nullptr, false, g_model.IsSelected(c))) { g_model.PushUndo(); g_model.CutSelection(); }
   320→        ImGui::Separator();
   321→        if (ImGui::MenuItem("\303\226ne Getir")) g_model.BringToFront(c);
   322→        if (ImGui::MenuItem("Arkaya G\303\266nder")) g_model.SendToBack(c);
   323→        ImGui::Separator();
   324→        if (ImGui::MenuItem("Sil (Del)")) { g_model.PushUndo(); g_model.SelectOnly(c); g_model.DeleteList(g_model.ValidSelection()); }
   325→        ImGui::EndPopup();
   326→    }
   327→    ImGui::PopID();
   328→    ImGui::PopID();
   329→}

   331→static void DrawCodetree(ImVec2 min, ImVec2 max) {
   332→    ImGui::SetNextWindowPos(min);
   333→    ImGui::SetNextWindowSize(ImVec2(max.x - min.x, max.y - min.y));
   334→    ImGui::Begin("Kod Gezgini", nullptr, ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoSavedSettings);
   335→    for (size_t i = 0; i < forms::Count(); i++) {
   336→        forms::FormEntry* f = forms::At(i);
   337→        ImGui::PushID((int)i);
   338→        ImGuiTreeNodeFlags fl = ImGuiTreeNodeFlags_DefaultOpen | ((int)i == forms::ActiveIndex() ? ImGuiTreeNodeFlags_Selected : 0);
   339→        bool open = ImGui::TreeNodeEx(f->name.c_str(), fl);
   340→        if (ImGui::IsItemClicked(0)) {
   341→            forms::SwitchTo((int)i);
   342→            g_model.SelectForm();      // forma tıkla = form seçimi
   343→        }
   344→        if (open) {
   345→            // objeler (hiyerarşik: konteyner çocukları ağaçta)
   346→            // AKTİF form → g_model (düzenleme yüzeyi); park edilmiş → girdi modeli
   347→            const Model& fm = ((int)i == forms::ActiveIndex()) ? g_model : f->model;
   348→            for (Ctrl* c : fm.ctrls)
   349→                if (!c->parent) DrawCtrlRow(c, (int)i, 0);
   350→            // form olayları
   351→            if (!f->model.formEvents.empty()) {
   352→                ImGui::Indent(18);
   353→                for (auto& ev : f->model.formEvents)
   354→                    ImGui::BulletText("%s \u2192 %s", (f->name + "_" + ev.first).c_str(), ev.second.c_str());
   355→                ImGui::Unindent(18);
   356→            }
   357→            ImGui::TreePop();
   358→        }
   359→        ImGui::PopID();
   360→    }
   361→    ImGui::End();
   362→}

   364→static void DrawFormList(ImVec2 min, ImVec2 max) {
   365→    ImGui::SetNextWindowPos(min);
   366→    ImGui::SetNextWindowSize(ImVec2(max.x - min.x, max.y - min.y));
   367→    ImGui::Begin("Formlar", nullptr, ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoSavedSettings);
   368→    for (size_t i = 0; i < forms::Count(); i++) {

   405→static void HandleShortcuts(ImGuiIO& io) {
   406→    // metin girişi odaktaysa kısayolları yeme (panellerdeki InputText)

   418→        else if (ImGui::IsKeyPressed(ImGuiKey_X, false)) {

   432→            }

   434→        else if (ImGui::IsKeyPressed(ImGuiKey_A, false)) {
   435→            // Ctrl+A — tümünü seç
   436→            g_model.SelectAll();
   437→            s_status = "T\303\274m\303\274 se\303\247ildi: " + std::to_string(g_model.selList.size()) + " nesne";
   438→        }
   439→        else if (ImGui::IsKeyPressed(ImGuiKey_S, false)) DoSave(false);
   440→        else if (ImGui::IsKeyPressed(ImGuiKey_O, false)) DoOpen();
   441→        else if (ImGui::IsKeyPressed(ImGuiKey_N, false)) DoNewProject();
   442→    } else {
   443→        if (ImGui::IsKeyPressed(ImGuiKey_F6, false)) s_genDialog = true;
   444→        else if (ImGui::IsKeyPressed(ImGuiKey_F5, false)) g_gridVisible = !g_gridVisible;
   445→        else if (ImGui::IsKeyPressed(ImGuiKey_Delete, false) && !g_model.selList.empty()) {
   446→            // Del — seçili NESNELERİ sil (form seçiliyken formu silmez)

   449→            g_model.DeleteList(g_model.ValidSelection());

   455→        }
   456→        else if (!g_model.selList.empty()) {
   457→            // ok tuşları: İnce ayar (Shift = ızgara adımı) — seçili GRUBU taşı
   458→            int step = io.KeyShift ? GRID_SIZE : 1;
   459→            int dx = 0, dy = 0;

   461→            if (ImGui::IsKeyPressed(ImGuiKey_RightArrow, true)) dx = step;
   462→            if (ImGui::IsKeyPressed(ImGuiKey_UpArrow, true)) dy = -step;
   463→            if (ImGui::IsKeyPressed(ImGuiKey_DownArrow, true)) dy = step;

   465→                for (Ctrl* c : g_model.ValidSelection()) { c->x += dx; c->y += dy; }
   466→                g_modelDirty = true;
   467→            }
   468→        }
   469→    }

   472→// ---------------------------------------------------------------------------
   473→//  selftest otomasyonu — Wine altında bağımsız test için
   474→//  Betik satırları:  add <tip> <x> <y> ; select <isim> ; set <isim> <prop> <değer>
   475→//  form <x> <y> ; newform ; save <yol> ; gen <yol> ; quit ; wait <kare>
   476→static int s_selftestFrame = 0;
   477→static std::vector<std::string> s_selftestLines;
   478→static bool s_selftestActive = false;
   479→static int s_selftestWaitUntil = -1;
   480→// olay enjeksiyon kuyruğu (karelere yayılmış mesajlar)
   481→struct InjectedMsg { int atFrame; UINT msg; WPARAM wp; LPARAM lp; };
   482→static std::vector<InjectedMsg> s_inject;
   483→static void Inject(int framesLater, UINT msg, WPARAM wp, LPARAM lp) {
   484→    InjectedMsg m; m.atFrame = s_selftestFrame + framesLater; m.msg = msg; m.wp = wp; m.lp = lp;
   485→    s_inject.push_back(m);
   486→}
   487→static void DrainInject() {
   488→    HWND hwnd = (HWND)g_mainHwnd;
   489→    if (!hwnd) return;
   490→    for (size_t i = 0; i < s_inject.size(); ) {
   491→        if (s_inject[i].atFrame <= s_selftestFrame) {
   492→            PostMessageA(hwnd, s_inject[i].msg, s_inject[i].wp, s_inject[i].lp);
   493→            s_inject.erase(s_inject.begin() + i);
   494→        } else i++;
   495→    }
   496→}
   497→static void SelftestStep() {
   498→    if (!s_selftestActive) return;
   499→    DrainInject();
   500→    if (s_selftestWaitUntil >= 0) {
   501→        if (s_selftestFrame < s_selftestWaitUntil) return;
   502→        s_selftestWaitUntil = -1;
   503→    }
   504→    while (!s_selftestLines.empty()) {
   505→        std::string line = TrimA(s_selftestLines.front());
   506→        s_selftestLines.erase(s_selftestLines.begin());
   507→        if (line.empty() || line[0] == '#') continue;
   508→        std::istringstream iss(line);
   509→        std::string cmd; iss >> cmd;
   510→        if (cmd == "add") {
   511→            std::string type; int x, y;
   512→            iss >> type >> x >> y;
   513→            Ctrl* c = g_model.Add(type, x, y, nullptr);
   514→            s_status = c ? ("Eklendi: " + c->name) : "Ekleme hatas\304\261";
   515→        } else if (cmd == "select") {
   516→            std::string n; iss >> n;
   517→            canvas::Select(g_model.FindByName(n));
   518→        } else if (cmd == "selectadd") {
   519→            std::string n; iss >> n;
   520→            Ctrl* c = g_model.FindByName(n);
   521→            if (c) g_model.AddSel(c);
   522→        } else if (cmd == "set") {
   523→            std::string n, p; iss >> n >> p;
   524→            std::string rest; std::getline(iss, rest);
   525→            rest = TrimA(rest);
   526→            // tırnak içi değeri soy: "#D25410" → #D25410
   527→            if (rest.size() >= 2 && (rest.front() == '"' || rest.front() == '\'') && rest.back() == rest.front())
   528→                rest = rest.substr(1, rest.size() - 2);
   529→            Ctrl* c = g_model.FindByName(n);
   530→            if (c) {
   531→                if (p == "Name") g_model.Rename(c, rest);
   532→                else c->props[p] = rest;
   533→            }
   534→        } else if (cmd == "form") { iss >> g_model.formW >> g_model.formH; g_model.formProps["ClientSize"] = std::to_string(g_model.formW) + ", " + std::to_string(g_model.formH); }
   535→        else if (cmd == "newform") { forms::NewForm(); }
   536→        else if (cmd == "save") { std::string p; std::getline(iss, p); p = TrimA(p); forms::SaveProjectAs(p); }
   537→        else if (cmd == "savefolder") { std::string p; std::getline(iss, p); p = TrimA(p); forms::SaveFolderProject(p); }
   538→        else if (cmd == "gen") { std::string p; std::getline(iss, p); p = TrimA(p); codegen::EmitProject(p); }
   539→        else if (cmd == "theme") { int t; iss >> t; theme::Apply(t); }
   540→        else if (cmd == "genbackend") { int b; iss >> b; codegen::g_settings.backend = b; }
   541→        else if (cmd == "projname") { std::string n; iss >> n; codegen::g_settings.projectName = n; }
   542→        else if (cmd == "wait") { int k; iss >> k; s_selftestWaitUntil = s_selftestFrame + k; return; }
   543→        else if (cmd == "arm") { std::string t; iss >> t; canvas::ArmTool(t); s_status = "ARM: " + t; }
   544→        else if (cmd == "click") {
   545→            // gerçek tıklama enjeksiyonu (ImGui Win32 backend'in işlediği WM mesajları)
   546→            int x, y; iss >> x >> y;
   547→            HWND hwnd = (HWND)g_mainHwnd;
   548→            if (hwnd) {
   549→                PostMessageA(hwnd, WM_MOUSEMOVE, 0, MAKELPARAM(x, y));
   550→                PostMessageA(hwnd, WM_LBUTTONDOWN, MK_LBUTTON, MAKELPARAM(x, y));
   551→                PostMessageA(hwnd, WM_LBUTTONUP, 0, MAKELPARAM(x, y));
   552→                s_status = "click " + std::to_string(x) + "," + std::to_string(y);
   553→            }
   554→        }
   555→        else if (cmd == "dblclick") {
   556→            int x, y; iss >> x >> y;
   557→            HWND hwnd = (HWND)g_mainHwnd;
   558→            if (hwnd) {
   559→                PostMessageA(hwnd, WM_MOUSEMOVE, 0, MAKELPARAM(x, y));
   560→                PostMessageA(hwnd, WM_LBUTTONDOWN, MK_LBUTTON, MAKELPARAM(x, y));
   561→                PostMessageA(hwnd, WM_LBUTTONUP, 0, MAKELPARAM(x, y));
   562→                PostMessageA(hwnd, WM_LBUTTONDBLCLK, MK_LBUTTON, MAKELPARAM(x, y));
   563→                PostMessageA(hwnd, WM_LBUTTONUP, 0, MAKELPARAM(x, y));
   564→            }
   565→        }
   566→        else if (cmd == "drag") {
   567→            // rubber band / nesne taşıma sürüklemesi — MESAJLAR KARELERE YAYILIR
   568→            // (tek karede down+up = tık olarak birleşir; sürükleme = ayrı kareler)
   569→            int x1, y1, x2, y2; iss >> x1 >> y1 >> x2 >> y2;
   570→            HWND hwnd = (HWND)g_mainHwnd;
   571→            if (hwnd) {
   572→                Inject(0, WM_MOUSEMOVE, 0, MAKELPARAM(x1, y1));
   573→                Inject(0, WM_LBUTTONDOWN, MK_LBUTTON, MAKELPARAM(x1, y1));
   574→                Inject(1, WM_MOUSEMOVE, MK_LBUTTON, MAKELPARAM(x1 + (x2 - x1) / 3, y1 + (y2 - y1) / 3));
   575→                Inject(2, WM_MOUSEMOVE, MK_LBUTTON, MAKELPARAM(x1 + 2 * (x2 - x1) / 3, y1 + 2 * (y2 - y1) / 3));
   576→                Inject(3, WM_MOUSEMOVE, MK_LBUTTON, MAKELPARAM(x2, y2));
   577→                Inject(4, WM_LBUTTONUP, 0, MAKELPARAM(x2, y2));
   578→            }
   579→        }
   580→        else if (cmd == "key") {
   581→            // key <ad> [ctrl] — ad: a-z, del, esc, left/right/up/down, enter, f5, f6
   582→            std::string k; int ctrl = 0; iss >> k >> ctrl;
   583→            UINT vk = 0;
   584→            if (k.size() == 1 && k[0] >= 'a' && k[0] <= 'z') vk = (UINT)(k[0] - 'a' + 'A');
   585→            else if (k == "del" || k == "delete") vk = VK_DELETE;
   586→            else if (k == "esc" || k == "escape") vk = VK_ESCAPE;
   587→            else if (k == "left") vk = VK_LEFT;
   588→            else if (k == "right") vk = VK_RIGHT;
   589→            else if (k == "up") vk = VK_UP;
   590→            else if (k == "down") vk = VK_DOWN;
   591→            else if (k == "enter") vk = VK_RETURN;
   592→            else if (k == "f5") vk = VK_F5;
   593→            else if (k == "f6") vk = VK_F6;
   594→            HWND hwnd = (HWND)g_mainHwnd;
   595→            if (hwnd && vk) {
   596→                if (ctrl) { PostMessageA(hwnd, WM_KEYDOWN, VK_CONTROL, 0x1D0001); }
   597→                PostMessageA(hwnd, WM_KEYDOWN, vk, 0);
   598→                PostMessageA(hwnd, WM_KEYUP, vk, 0);
   599→                if (ctrl) { PostMessageA(hwnd, WM_KEYUP, VK_CONTROL, 0x1D0001); }
   600→            }
   601→        }
   602→        else if (cmd == "paste") { g_model.PasteClipboard(GRID_SIZE, GRID_SIZE); }
   603→        else if (cmd == "copy") { g_model.CopySelection(); }
   604→        else if (cmd == "cut") { g_model.CutSelection(); }
   605→        else if (cmd == "selectall") { g_model.SelectAll(); }
   606→        else if (cmd == "selcount") {
   607→            // seçim sayısını status'a yaz — test doğrulaması
   608→            s_status = "SEL:" + std::to_string(g_model.selList.size()) +
   609→                       " CLIP:" + std::to_string(!g_model.clipboardJson.empty());
   610→        }
   611→        else if (cmd == "quit") { PostQuitMessage(0); s_selftestActive = false; return; }
   612→    }
   613→    s_selftestActive = false;
   614→}
