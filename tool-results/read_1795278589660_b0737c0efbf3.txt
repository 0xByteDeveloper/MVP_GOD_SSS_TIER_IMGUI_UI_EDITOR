    36→ImVec2 CtrlAbsPos(const Ctrl* c) {
    37→    int x = c->x, y = c->y;
    38→    for (Ctrl* p = c->parent; p; p = p->parent) { x += p->x; y += p->y; }
    39→    return ImVec2((float)x, (float)y);
    40→}

   268→void ClickSelectUnder(ImVec2 pt) {
   269→    // en üstteki (son çizilen) eşleşme
   270→    for (int i = (int)s_hitRects.size() - 1; i >= 0; i--) {
   271→        WidgetRect& r = s_hitRects[i];
   272→        if (pt.x >= r.rmin.x && pt.x <= r.rmax.x && pt.y >= r.rmin.y && pt.y <= r.rmax.y) {
   273→            Select(r.c);
   274→            return;
   275→        }
   276→    }
   277→    SelectForm();
   278→}

   280→static bool HandleHit(ImVec2 pt, ImVec2 rmin, ImVec2 rmax, float hs) {
   281→    // 0:sol üst 1:üst-orta 2:sağ üst 3:sağ-orta 4:sağ alt 5:alt-orta 6:sol alt 7:sol-orta
   282→    float l = rmin.x, t = rmin.y, r = rmax.x, b = rmax.y;
   283→    ImVec2 c[8] = { {l,t},{(l+r)/2,t},{r,t},{r,(t+b)/2},{r,b},{(l+r)/2,b},{l,b},{l,(t+b)/2} };
   284→    for (int i = 0; i < 8; i++) {
   285→        if (fabsf(pt.x - c[i].x) <= hs && fabsf(pt.y - c[i].y) <= hs) return true;
   286→    }
   287→    return false;
   288→}

   290→static int HandleIndexAt(ImVec2 pt, ImVec2 rmin, ImVec2 rmax, float hs) {
   291→    float l = rmin.x, t = rmin.y, r = rmax.x, b = rmax.y;
   292→    ImVec2 c[8] = { {l,t},{(l+r)/2,t},{r,t},{r,(t+b)/2},{r,b},{(l+r)/2,b},{l,b},{l,(t+b)/2} };
   293→    for (int i = 0; i < 8; i++)
   294→        if (fabsf(pt.x - c[i].x) <= hs && fabsf(pt.y - c[i].y) <= hs) return i;
   295→    return -1;
   296→}

   300→void Draw(const ImVec2& rectMin, const ImVec2& rectMax) {
   301→    s_canvasMin = rectMin; s_canvasMax = rectMax;
   302→    s_hitRects.clear();
   303→    Model& m = g_model;
   304→
   305→    // canvas zemin
   306→    ImDrawList* dl = ImGui::GetWindowDrawList();
   307→    dl->AddRectFilled(rectMin, rectMax, IM_COL32(38, 40, 48, 255));
   308→    // ızgara
   309→    if (g_gridVisible) {
   310→        ImU32 dot = IM_COL32(90, 94, 110, 200);
   311→        for (float x = rectMin.x; x < rectMax.x; x += GRID_SIZE)
   312→            for (float y = rectMin.y; y < rectMax.y; y += GRID_SIZE)
   313→                dl->AddCircleFilled(ImVec2(x, y), 0.7f, dot, 4);
   314→    }
   315→
   316→    // tasarlanan form — GERÇEK ImGui penceresi (canvas alanıyla kısıtlanmış)
   317→    std::string title = m.formProps.count("Text") ? m.formProps["Text"] : "Form1";
   318→    ImGuiWindowFlags flags = ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoSavedSettings;
   319→    // Flags prop'ları üretilen kodla aynı set
   320→    auto it = m.formProps.find("Flags");
   321→    if (it != m.formProps.end())
   322→        for (auto& s : SplitW(Utf8ToWideStr(it->second), L'|')) {
   323→            std::string t = TrimA(WideStrToUtf8(s));
   324→            if (t == "NoTitleBar") flags |= ImGuiWindowFlags_NoTitleBar;
   325→            else if (t == "NoResize") flags |= ImGuiWindowFlags_NoResize;
   326→            else if (t == "NoCollapse") flags |= ImGuiWindowFlags_NoCollapse;
   327→            else if (t == "NoMove") flags |= ImGuiWindowFlags_NoMove;
   328→            else if (t == "AlwaysAutoResize") flags |= ImGuiWindowFlags_AlwaysAutoResize;
   329→            else if (t == "NoScrollbar") flags |= ImGuiWindowFlags_NoScrollbar;
   330→            else if (t == "MenuBar") flags |= ImGuiWindowFlags_MenuBar;
   331→        }
   332→    // form menüsü varsa MenuBar bayrağı
   333→    for (Ctrl* c : m.ctrls) if (c->type == "Menu") { flags |= ImGuiWindowFlags_MenuBar; break; }
   334→
   335→    ImVec2 formPos(rectMin.x + 40.0f, rectMin.y + 30.0f);
   336→    ImGui::SetNextWindowPos(formPos);
   337→    ImGui::SetNextWindowSize(ImVec2((float)m.formW, (float)m.formH));
   338→    ImGui::PushStyleColor(ImGuiCol_WindowBg, IM_COL32(48, 50, 58, 255));   // tasarım zemini hafif farklı
   339→    bool open = ImGui::Begin(title.c_str(), nullptr, flags);
   340→    ImGui::PopStyleColor();
   341→    if (open) {
   342→        // menü (varsa) — üretilen kodla aynı Menus/Commands mantığı
   343→        for (Ctrl* c : m.ctrls) if (c->type == "Menu") {
   344→            if (ImGui::BeginMenuBar()) {
   345→                for (auto& s : SplitW(Utf8ToWideStr(c->Get("Items", "Dosya|Düzen|Yardım")), L'|')) {
   346→                    std::string men = WideStrToUtf8(s);
   347→                    if (ImGui::BeginMenu(men.c_str())) {
   348→                        for (auto& cmd : SplitW(Utf8ToWideStr(c->Get("Commands", "")), L'|')) {
   349→                            std::string cline = WideStrToUtf8(cmd);
   350→                            size_t col = cline.find(':');
   351→                            if (col == std::string::npos) continue;
   352→                            std::string owner = TrimA(cline.substr(0, col));
   353→                            std::string item = TrimA(cline.substr(col + 1));
   354→                            if (owner == men && !item.empty()) ImGui::MenuItem(item.c_str());
   355→                        }
   356→                        ImGui::EndMenu();
   357→                    }
   358→                }
   359→                ImGui::EndMenuBar();
   360→            }
   361→            // menü çerçevesini seçim dikdörtgeni olarak ekle
   362→        }
   363→        // üst-seviye nesneler (parent=null) — z sırası listedeki sıra

   365→        for (Ctrl* c : m.ctrls) if (!c->parent) tops.push_back(c);

   368→    ImGui::End();
   369→
   370→    // bileşen tepsisi (görsel olmayan) — form altında
   371→    {
   372→        float trayY = formPos.y + m.formH + 14.0f;
   373→        float trayX = formPos.x;

   375→        dl->AddText(ImVec2(trayX + 6, trayY + 6), IM_COL32(150, 152, 160, 255), "Bile\305\237enler:");

   377→        for (Ctrl* c : m.ctrls) {
   378→            if (c->type == "Timer" || c->type == "FontRef") {
   379→                std::string lbl = "[ " + c->name + " ]";
   380→                ImVec2 ts = ImGui::CalcTextSize(lbl.c_str());
   381→                dl->AddRectFilled(ImVec2(cx, trayY + 3), ImVec2(cx + ts.x + 10, trayY + 23), IM_COL32(55, 58, 68, 255), 3.0f);
   382→                dl->AddText(ImVec2(cx + 5, trayY + 6), IM_COL32(200, 202, 210, 255), lbl.c_str());
   383→                // tıklanabilir alan kaydı
   384→                WidgetRect r; r.c = c;
   385→                r.rmin = ImVec2(cx, trayY + 3); r.rmax = ImVec2(cx + ts.x + 10, trayY + 23);
   386→                s_hitRects.push_back(r);
   387→                cx += ts.x + 16;

   389→        }
   390→    }
   391→
   392→    // ---- seçim vurgusu + tutamaçlar ----
   393→    ImGuiIO& io = ImGui::GetIO();
   394→    if (s_sel || g_model.formSelected) {
   395→        ImVec2 rmin, rmax;
   396→        if (s_sel) {
   397→            // seçilen nesnenin bu karedeki gerçek çizim dikdörtgenini bul
   398→            bool found = false;
   399→            for (auto& r : s_hitRects) if (r.c == s_sel) { rmin = r.rmin; rmax = r.rmax; found = true; break; }
   400→            if (!found) { rmin = formPos + CtrlAbsPos(s_sel); rmax = rmin; }   // kaba yedek
   401→        } else {
   402→            rmin = ImGui::GetItemRectMin();   // form: son End() — boşsa form kendisi
   403→            rmin = formPos;
   404→            rmax = ImVec2(formPos.x + m.formW, formPos.y + m.formH);
   405→        }
   406→        ImU32 accent = IM_COL32(255, 176, 32, 255);
   407→        dl->AddRect(ImVec2(rmin.x - 2, rmin.y - 2), ImVec2(rmax.x + 2, rmax.y + 2), accent, 0.0f, 0, 1.5f);
   408→        float hs = 4.5f;
   409→        ImVec2 c[8] = { {rmin.x,rmin.y},{(rmin.x+rmax.x)/2,rmin.y},{rmax.x,rmin.y},{rmax.x,(rmin.y+rmax.y)/2},
   410→                        {rmax.x,rmax.y},{(rmin.x+rmax.x)/2,rmax.y},{rmin.x,rmax.y},{rmin.x,(rmin.y+rmax.y)/2} };
   411→        for (int i = 0; i < 8; i++) {
   412→            dl->AddRectFilled(ImVec2(c[i].x - hs, c[i].y - hs), ImVec2(c[i].x + hs, c[i].y + hs), accent);
   413→            dl->AddRect(ImVec2(c[i].x - hs, c[i].y - hs), ImVec2(c[i].x + hs, c[i].y + hs), IM_COL32(0, 0, 0, 255));

   415→
   416→        // ---- etkileşim: taşıma / boyutlandırma ----
   417→        ImVec2 mp = io.MousePos;
   418→        bool hovered = mp.x >= rmin.x - SEL_MARGIN && mp.x <= rmax.x + SEL_MARGIN && mp.y >= rmin.y - SEL_MARGIN && mp.y <= rmax.y + SEL_MARGIN;
   419→
   420→        if (ImGui::IsMouseClicked(0) && hovered) {
   421→            int hi = HandleIndexAt(mp, rmin, rmax, hs + 2.5f);
   422→            if (s_sel && hi >= 0) {
   423→                s_dragHandle = hi;
   424→                g_model.PushUndo();
   425→                s_origX = s_sel->x; s_origY = s_sel->y; s_origW = s_sel->w; s_origH = s_sel->h;
   426→                s_dragStart = mp;
   427→            } else {
   428→                s_dragBody = true;
   429→                if (s_sel) {
   430→                    g_model.PushUndo();
   431→                    s_origX = s_sel->x; s_origY = s_sel->y;
   432→                } else {
   433→                    g_model.PushUndo();
   434→                    s_origX = m.formW; s_origY = m.formH;
   435→                }
   436→                s_dragStart = mp;
   437→            }
   438→        }
   439→        if (s_dragBody && ImGui::IsMouseReleased(0)) s_dragBody = false;
   440→        if (s_dragHandle >= 0 && ImGui::IsMouseReleased(0)) s_dragHandle = -1;
   441→
   442→        if (s_dragBody && ImGui::IsMouseDown(0)) {
   443→            ImVec2 d(mp.x - s_dragStart.x, mp.y - s_dragStart.y);
   444→            if (s_sel) {
   445→                int nx = s_origX + (int)d.x, ny = s_origY + (int)d.y;
   446→                if (g_snapEnabled) { nx = (nx + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE; ny = (ny + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE; }
   447→                s_sel->x = nx; s_sel->y = ny;
   448→                g_modelDirty = true;
   449→            } else {
   450→                int nw = s_origX + (int)d.x, nh = s_origY + (int)d.y;
   451→                if (g_snapEnabled) { nw = (nw + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE; nh = (nh + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE; }
   452→                m.formW = ImMax(nw, 120); m.formH = ImMax(nh, 80);
   453→                m.formProps["ClientSize"] = std::to_string(m.formW) + ", " + std::to_string(m.formH);
   454→                g_modelDirty = true;
   455→            }
   456→        }
   457→        if (s_dragHandle >= 0 && ImGui::IsMouseDown(0) && s_sel) {
   458→            ImVec2 d(mp.x - s_dragStart.x, mp.y - s_dragStart.y);
   459→            int nx = s_sel->x, ny = s_sel->y, nw = s_origW, nh = s_origH;
   460→            if (s_dragHandle == 0 || s_dragHandle == 6 || s_dragHandle == 7) { nw = s_origW - (int)d.x; nx = s_origX + (int)d.x; }
   461→            if (s_dragHandle == 0 || s_dragHandle == 1 || s_dragHandle == 2) { nh = s_origH - (int)d.y; ny = s_origY + (int)d.y; }
   462→            if (s_dragHandle == 2 || s_dragHandle == 3 || s_dragHandle == 4) { nw = s_origW + (int)d.x; }
   463→            if (s_dragHandle == 4 || s_dragHandle == 5 || s_dragHandle == 6) { nh = s_origH + (int)d.y; }
   464→            nw = ImMax(nw, 12); nh = ImMax(nh, 8);
   465→            if (g_snapEnabled) {
   466→                nx = (nx + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE;
   467→                ny = (ny + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE;
   468→                nw = (nw + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE;
   469→                nh = (nh + GRID_SIZE/2) / GRID_SIZE * GRID_SIZE;

   477→    if (!g_armTool.empty()) {
   478→        dl->AddRect(rectMin, rectMax, IM_COL32(90, 200, 120, 200), 4.0f, 0, 2.0f);
   479→        if (ImGui::IsMouseClicked(0) && !io.WantCaptureKeyboard) {
   480→            ImVec2 pt = io.MousePos;
   481→            if (pt.x > rectMin.x && pt.x < rectMax.x && pt.y > rectMin.y && pt.y < rectMax.y) {

   484→                // tıklanan nokta bir konteynerin içinde mi? (basit: çizilen hitRects'ten üstteki konteyner)
   485→                Ctrl* parent = nullptr;
   486→                for (int i = (int)s_hitRects.size() - 1; i >= 0; i--) {

   492→                            px = (int)(pt.x - r.rmin.x); py = (int)(pt.y - r.rmin.y);

   494→                        }
   495→                    }

   497→                Ctrl* added = g_model.Add(g_armTool, px, py, parent);

   499→                    // konteyner çocuğu olarak parent'a eklendi — parent'ın çocuk listesine push edildi (model.Add parent'ı set eder)
   500→                }
   501→                g_armTool.clear();
   502→                g_armParent = nullptr;
   503→            } else {
   504→                g_armTool.clear();   // canvas dışı tık = iptal
   505→            }
   506→        }
   507→    }

   509→    // seçim tıklaması (drag değilken + yerleştirme yokken) — canvas boş alanı
   510→    if (!s_dragBody && s_dragHandle < 0 && g_armTool.empty() && ImGui::IsMouseClicked(0) &&
   511→        io.MousePos.x > rectMin.x && io.MousePos.x < rectMax.x && io.MousePos.y > rectMin.y && io.MousePos.y < rectMax.y) {

   514→            // canvas arka planına tıklandı → form seçimi (ya da alttaki nesne)
   515→            ClickSelectUnder(io.MousePos);
   516→        } else {

   518→            ClickSelectUnder(io.MousePos);
   519→        }
   520→    }

   525→}
