    36→std::string OpenDlg(const char* title, const char* filter) {
    37→    char file[MAX_PATH] = "";
    38→    OPENFILENAMEA ofn = { sizeof(ofn) };
    39→    ofn.hwndOwner = GetActiveWindow();
    40→    ofn.lpstrFile = file;
    41→    ofn.nMaxFile = MAX_PATH;
    42→    ofn.lpstrTitle = title;
    43→    ofn.lpstrFilter = filter;
    44→    ofn.Flags = OFN_FILEMUSTEXIST | OFN_EXPLORER | OFN_NOCHANGEDIR;
    45→    if (GetOpenFileNameA(&ofn)) return file;
    46→    return "";
    47→}
    48→std::string SaveDlg(const char* title, const char* filter, const char* defExt) {
    49→    char file[MAX_PATH] = "";
    50→    OPENFILENAMEA ofn = { sizeof(ofn) };
    51→    ofn.hwndOwner = GetActiveWindow();
    52→    ofn.lpstrFile = file;
    53→    ofn.nMaxFile = MAX_PATH;
    54→    ofn.lpstrTitle = title;
    55→    ofn.lpstrFilter = filter;
    56→    ofn.lpstrDefExt = defExt;
    57→    ofn.Flags = OFN_OVERWRITEPROMPT | OFN_EXPLORER | OFN_NOCHANGEDIR;
    58→    if (GetSaveFileNameA(&ofn)) return file;
    59→    return "";
    60→}

   176→static void DoSave(bool saveAs) {
   177→    if (saveAs) {
   178→        std::string p = SaveDlg("Proje kaydet", "IMGUI GOD projesi\0*.imgod\0", "imgod");
   179→        if (p.empty()) return;
   180→        if (forms::SaveProjectAs(p)) s_status = "Kaydedildi: " + p;
   181→        else s_status = "Kaydedilemedi";
   182→        return;
   183→    }
   184→    if (!forms::ProjectDir().empty()) {
   185→        if (forms::SaveFolderProject(forms::ProjectDir())) s_status = "Kaydedildi: " + forms::ProjectDir();
   186→        else s_status = "Kaydedilemedi";
   187→    } else DoSave(true);
   188→}

   193→    else s_status = "Kaydedilemedi";

   242→static void DrawToolbox(ImVec2 min, ImVec2 max) {
   243→    ImGui::SetNextWindowPos(min);
   244→    ImGui::SetNextWindowSize(ImVec2(max.x - min.x, max.y - min.y));
   245→    ImGui::Begin("Ara\303\247 Kutusu", nullptr, ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoSavedSettings);
   246→    static char search[128] = "";
   247→    ImGui::TextUnformatted("Ara (kategori/nesne):");
   248→    ImGui::SetNextItemWidth(-1);
   249→    ImGui::InputText("##tbs", search, sizeof search);
   250→    ImGui::Separator();
   251→    for (auto& cat : g_catalog.categories) {
   252→        // arama filtresi
   253→        bool show = search[0] == 0;
   254→        for (auto& e : g_catalog.catElements[cat]) {
   255→            std::wstring low = LowerW(Utf8ToWideStr(e));
   256→            std::wstring sw = LowerW(Utf8ToWideStr(std::string(search)));
   257→            if (low.find(sw) != std::wstring::npos || cat.find(search) != std::string::npos) show = true;
   258→        }
   259→        if (!show) continue;
   260→        ImGui::SetNextItemOpen(true, ImGuiCond_Once);
   261→        if (ImGui::TreeNode(cat.c_str())) {
   262→            for (auto& e : g_catalog.catElements[cat]) {
   263→                ImGui::PushID(e.c_str());
   264→                if (ImGui::Selectable(e.c_str(), canvas::g_armTool == e)) {
   265→                    canvas::g_armTool = e;   // yerleştirme moduna gir
   266→                    s_status = "Yerle\305\237tir: " + e + " — canvas'a t\304\261kla";
   267→                }
   268→                if (ImGui::IsItemHovered()) {
   269→                    const ElementDef* d = g_catalog.Find(e);
   270→                    if (d) ImGui::SetTooltip("%s\n%s\n\303\227ift t\304\261kla yerle\305\237tir, canvas'ta konum se\303\247",
   271→                                             d->cppApi.c_str(), d->kind.c_str());
   272→                }
   273→                ImGui::PopID();
   274→            }
   275→            ImGui::TreePop();
   276→        }
   277→    }
   278→    ImGui::End();
   279→}

   281→static void DrawCodetree(ImVec2 min, ImVec2 max) {
   282→    ImGui::SetNextWindowPos(min);
   283→    ImGui::SetNextWindowSize(ImVec2(max.x - min.x, max.y - min.y));
   284→    ImGui::Begin("Kod Gezgini", nullptr, ImGuiWindowFlags_NoCollapse | ImGuiWindowFlags_NoMove | ImGuiWindowFlags_NoResize | ImGuiWindowFlags_NoSavedSettings);
   285→    for (size_t i = 0; i < forms::Count(); i++) {

   287→        ImGui::PushID((int)i);

   289→        bool open = ImGui::TreeNodeEx(f->name.c_str(), fl);
   290→        if (ImGui::IsItemClicked()) forms::SwitchTo((int)i);
   291→        if (open) {
   292→            // objeler
   293→            for (Ctrl* c : f->model.ctrls) {
   294→                ImGui::BulletText("%s : %s", c->name.c_str(), c->type.c_str());
   295→                if (ImGui::IsItemClicked()) {
   296→                    if ((int)i != forms::ActiveIndex()) forms::SwitchTo((int)i);
   297→                    canvas::Select(c);
   298→                }
   299→                if (ImGui::IsItemHovered()) ImGui::SetTooltip("Tek t\304\261kla: se\303\247");
   300→                // olayları iç içe
   301→                if (!c->events.empty()) {
   302→                    ImGui::Indent(18);
   303→                    for (auto& ev : c->events) ImGui::BulletText("%s \u2192 %s", ev.first.c_str(), ev.second.c_str());
   304→                    ImGui::Unindent(18);
   305→                }
   306→            }
   307→            // form olayları
   308→            for (auto& ev : f->model.formEvents)
   309→                ImGui::BulletText("%s \u2192 %s", (f->name + "_" + ev.first).c_str(), ev.second.c_str());
   310→            ImGui::TreePop();
   311→        }
   312→        ImGui::PopID();
   313→    }
   314→    ImGui::End();

   357→static void HandleShortcuts(ImGuiIO& io) {
   358→    if (io.KeyCtrl) {
   359→        if (ImGui::IsKeyPressed(ImGuiKey_Z, false)) { g_model.Undo(); s_status = "Geri al\304\261nd\304\261"; }
   360→        else if (ImGui::IsKeyPressed(ImGuiKey_Y, false)) { g_model.Redo(); s_status = "Yinelendi"; }
   361→        else if (ImGui::IsKeyPressed(ImGuiKey_C, false) && g_model.selected) {
   362→            // Ctrl+C — tek nesne JSON kopyala
   363→            std::ostringstream o;
   364→            o << "{";
   365→            Ctrl* c = g_model.selected;
   366→            o << "\"Name\":\"" << c->name << "\",\"Type\":\"" << c->type << "\",\"X\":" << c->x << ",\"Y\":" << c->y;
   367→            o << ",\"W\":" << c->w << ",\"H\":" << c->h << ",\"Props\":{";
   368→            bool first = true;
   369→            for (auto& kv : c->props) {
   370→                if (!first) o << ",";
   371→                o << "\"" << kv.first << "\":\"" << kv.second << "\"";
   372→                first = false;
   373→            }
   374→            o << "}}";
   375→            g_clipboardJson = o.str();
   376→            s_status = "Kopyaland\304\261: " + c->name;
   377→        }

   409→// ---------------------------------------------------------------------------
   410→//  selftest otomasyonu — Wine altında bağımsız test için
   411→//  Betik satırları:  add <tip> <x> <y> ; select <isim> ; set <isim> <prop> <değer>
   412→//  form <x> <y> ; newform ; save <yol> ; gen <yol> ; quit ; wait <kare>
   413→static int s_selftestFrame = 0;
   414→static std::vector<std::string> s_selftestLines;
   415→static bool s_selftestActive = false;
   416→static int s_selftestWaitUntil = -1;
   417→static void SelftestStep() {
   418→    if (!s_selftestActive) return;
   419→    if (s_selftestWaitUntil >= 0) {
   420→        if (s_selftestFrame < s_selftestWaitUntil) return;
   421→        s_selftestWaitUntil = -1;
   422→    }
   423→    while (!s_selftestLines.empty()) {
   424→        std::string line = TrimA(s_selftestLines.front());
   425→        s_selftestLines.erase(s_selftestLines.begin());
   426→        if (line.empty() || line[0] == '#') continue;
   427→        std::istringstream iss(line);
   428→        std::string cmd; iss >> cmd;
   429→        if (cmd == "add") {
   430→            std::string type; int x, y;
   431→            iss >> type >> x >> y;
   432→            Ctrl* c = g_model.Add(type, x, y, nullptr);
   433→            s_status = c ? ("Eklendi: " + c->name) : "Ekleme hatas\304\261";
   434→        } else if (cmd == "select") {
   435→            std::string n; iss >> n;
   436→            canvas::Select(g_model.FindByName(n));
   437→        } else if (cmd == "set") {
   438→            std::string n, p; iss >> n >> p;
   439→            std::string rest; std::getline(iss, rest);
   440→            rest = TrimA(rest);
   441→            Ctrl* c = g_model.FindByName(n);
   442→            if (c) {
   443→                if (p == "Name") g_model.Rename(c, rest);
   444→                else c->props[p] = rest;
   445→            }
   446→        } else if (cmd == "form") { iss >> g_model.formW >> g_model.formH; g_model.formProps["ClientSize"] = std::to_string(g_model.formW) + ", " + std::to_string(g_model.formH); }
   447→        else if (cmd == "newform") { forms::NewForm(); }
   448→        else if (cmd == "save") { std::string p; std::getline(iss, p); p = TrimA(p); forms::SaveProjectAs(p); }
   449→        else if (cmd == "savefolder") { std::string p; std::getline(iss, p); p = TrimA(p); forms::SaveFolderProject(p); }
   450→        else if (cmd == "gen") { std::string p; std::getline(iss, p); p = TrimA(p); codegen::EmitProject(p); }
   451→        else if (cmd == "theme") { int t; iss >> t; theme::Apply(t); }
   452→        else if (cmd == "genbackend") { int b; iss >> b; codegen::g_settings.backend = b; }
   453→        else if (cmd == "wait") { int k; iss >> k; s_selftestWaitUntil = s_selftestFrame + k; return; }
   454→        else if (cmd == "quit") { PostQuitMessage(0); s_selftestActive = false; return; }
   455→    }
   456→    s_selftestActive = false;
   457→}
