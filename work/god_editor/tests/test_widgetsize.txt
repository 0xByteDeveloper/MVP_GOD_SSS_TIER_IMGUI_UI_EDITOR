# WIDGET BOYUTLANDIRMA TESTİ — kullanıcı hatası:
# "combo/listbox gibi şeyler görsel olarak küçülüp büyümüyo ama özelliklerdeki
#  Boyut değişiyo" → her tip artık GERÇEK çizim boyutuyla WYSIWYG.
# Nesne adları tipin İLK HARFİ küçültülmüş halidir: listBox1, sliderFloat1...
snap off
wait 15
add Combo 40 40
add ListBox 40 120
add SliderFloat 40 260
add InputText 40 320
add Checkbox 40 380
add Table 260 40
wait 25
# ---- taban ölçüleri (font metriğinden bağımsız DELTA doğrulaması için) ----
objsize combo1 state_wsz.txt
objsize listBox1 state_wsz.txt
objsize sliderFloat1 state_wsz.txt
objsize inputText1 state_wsz.txt
objsize checkbox1 state_wsz.txt
objsize table1 state_wsz.txt
# 1) Combo: genişlik +120 → GÖRSEL kutu +120 (etiket dışarıda, delta sabit)
set combo1 Width 320
wait 12
objsize combo1 state_wsz.txt
# 2) ListBox: TAM boyut (w,h) görselde (BeginListBox size — etiket yok, mutlak)
set listBox1 Width 260
set listBox1 Height 180
wait 12
objsize listBox1 state_wsz.txt
# 3) SliderFloat genişliği +100 (etiket çerçeve İÇİNDE → delta = mutlak)
set sliderFloat1 Width 300
wait 12
objsize sliderFloat1 state_wsz.txt
# 4) InputText genişliği +50 (etiket dışarıda — delta)
set inputText1 Width 250
wait 12
objsize inputText1 state_wsz.txt
# 5) Checkbox (5. TUR): Width 500 → kontrol alanı 500'de KALIR + yükseklik
#    ölçeklenince kare büyür (FULL boyutlandırma — kullanıcı talebi)
set checkbox1 Width 500
set checkbox1 Height 36
wait 12
objsize checkbox1 state_wsz.txt
# 6) Table genişliği görselde (outer_size.x — etiket yok, mutlak)
set table1 Width 360
wait 12
objsize table1 state_wsz.txt
# 7) SAĞ KENAR TUTAMACI sürüklemesi: combo +80 (W-maskesi + ApplyHandle)
select combo1
dragobj combo1 80 0
wait 12
objsize combo1 state_wsz.txt
# model boyutları kalıcı olarak kaydedilir
save state_wsize.imgod
wait 5
quit
